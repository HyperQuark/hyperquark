7pub fn wasm(func: &StepFunc, _inputs: Rc<[IrType]>) -> HQResult<Vec<InternalInstruction>> {
8    let ir_target_index: i32 = func
9        .target_index()
10        .try_into()
11        .map_err(|_| make_hq_bug!("target index out of bounds"))?;
12    let move_func_index = func.registries().external_functions().register(
13        ("motion", "gotoxy".into()),
14        (vec![ValType::F64, ValType::F64, ValType::I32], vec![]),
15    )?;
16    let pen_func_index = func.registries().external_functions().register(
17        ("pen", "line".into()),
18        (
19            vec![
20                ValType::F64,
21                ValType::F64,
22                ValType::F64,
23                ValType::F64,
24                ValType::F64,
25                ValType::F32,
26                ValType::F32,
27                ValType::F32,
28                ValType::F32,
29            ],
30            vec![],
31        ),
32    )?;
33    let StepTarget::Sprite(wasm_target_index) = func.target() else {
34        hq_bad_proj!("motion_sety called in stage")
35    };
36    let x_local = func.local(ValType::F64)?;
37    let y_local = func.local(ValType::F64)?;
38    func.free_local(x_local)?;
39    func.free_local(y_local)?;
40    Ok(wasm![
41        LocalSet(y_local),
42        I32Const(0),
43        F64Load(MemArg {
44            offset: (stage_layout::BLOCK_SIZE
45                + wasm_target_index * sprite_layout::BLOCK_SIZE
46                + sprite_layout::X)
47                .into(),
48            align: 3,
49            memory_index: 0
50        }),
51        LocalSet(x_local),
52        I32Const(0),
53        I32Load8U(MemArg {
54            offset: (stage_layout::BLOCK_SIZE
55                + wasm_target_index * sprite_layout::BLOCK_SIZE
56                + sprite_layout::PEN_DOWN)
57                .into(),
58            align: 0,
59            memory_index: 0
60        }),
61        If(wasm_encoder::BlockType::Empty),
62        I32Const(0),
63        F64Load(MemArg {
64            offset: (stage_layout::BLOCK_SIZE
65                + wasm_target_index * sprite_layout::BLOCK_SIZE
66                + sprite_layout::PEN_SIZE)
67                .into(),
68            align: 3,
69            memory_index: 0
70        }),
71        I32Const(0),
72        F64Load(MemArg {
73            offset: (stage_layout::BLOCK_SIZE
74                + wasm_target_index * sprite_layout::BLOCK_SIZE
75                + sprite_layout::X)
76                .into(),
77            align: 3,
78            memory_index: 0
79        }),
80        I32Const(0),
81        F64Load(MemArg {
82            offset: (stage_layout::BLOCK_SIZE
83                + wasm_target_index * sprite_layout::BLOCK_SIZE
84                + sprite_layout::Y)
85                .into(),
86            align: 3,
87            memory_index: 0
88        }),
89        LocalGet(x_local),
90        LocalGet(y_local),
91        I32Const(0),
92        F32Load(MemArg {
93            offset: (stage_layout::BLOCK_SIZE
94                + wasm_target_index * sprite_layout::BLOCK_SIZE
95                + sprite_layout::PEN_COLOR_R)
96                .into(),
97            align: 2,
98            memory_index: 0
99        }),
100        I32Const(0),
101        F32Load(MemArg {
102            offset: (stage_layout::BLOCK_SIZE
103                + wasm_target_index * sprite_layout::BLOCK_SIZE
104                + sprite_layout::PEN_COLOR_G)
105                .into(),
106            align: 2,
107            memory_index: 0
108        }),
109        I32Const(0),
110        F32Load(MemArg {
111            offset: (stage_layout::BLOCK_SIZE
112                + wasm_target_index * sprite_layout::BLOCK_SIZE
113                + sprite_layout::PEN_COLOR_B)
114                .into(),
115            align: 2,
116            memory_index: 0
117        }),
118        I32Const(0),
119        F32Load(MemArg {
120            offset: (stage_layout::BLOCK_SIZE
121                + wasm_target_index * sprite_layout::BLOCK_SIZE
122                + sprite_layout::PEN_COLOR_A)
123                .into(),
124            align: 2,
125            memory_index: 0
126        }),
127        Call(pen_func_index),
128        End,
129        I32Const(0),
130        LocalGet(x_local),
131        F64Store(MemArg {
132            offset: (stage_layout::BLOCK_SIZE
133                + wasm_target_index * sprite_layout::BLOCK_SIZE
134                + sprite_layout::X)
135                .into(),
136            align: 3,
137            memory_index: 0,
138        }),
139        I32Const(0),
140        LocalGet(y_local),
141        F64Store(MemArg {
142            offset: (stage_layout::BLOCK_SIZE
143                + wasm_target_index * sprite_layout::BLOCK_SIZE
144                + sprite_layout::Y)
145                .into(),
146            align: 3,
147            memory_index: 0,
148        }),
149        LocalGet(x_local),
150        LocalGet(y_local),
151        I32Const(ir_target_index),
152        Call(move_func_index),
153    ])
154}

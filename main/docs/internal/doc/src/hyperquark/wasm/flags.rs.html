99#[derive(Clone, Serialize, Deserialize)]
100#[wasm_bindgen(getter_with_clone)]
101#[expect(
102    clippy::unsafe_derive_deserialize,
103    reason = "wasm-bindgen introduces unsafe methods"
104)]
105pub struct FlagInfo {
106    /// a human-readable name for the flag
107    pub name: String,
108    pub description: String,
109    pub ty: String,
110    /// which WASM features does this flag rely on?
111    wasm_features: BTreeMap<String, Vec<WasmFeature>>,
112}
113
114#[wasm_bindgen]
115impl FlagInfo {
116    fn new() -> Self {
117        Self {
118            name: String::new(),
119            description: String::new(),
120            ty: String::new(),
121            wasm_features: BTreeMap::default(),
122        }
123    }
124
125    fn with_name(mut self, name: &str) -> Self {
126        self.name = name.to_string();
127        self
128    }
129
130    fn with_description(mut self, description: &str) -> Self {
131        self.description = description.to_string();
132        self
133    }
134
135    fn with_ty(mut self, ty: &str) -> Self {
136        self.ty = ty.to_string();
137        self
138    }
139
140    fn with_wasm_features(mut self, wasm_features: BTreeMap<String, Vec<WasmFeature>>) -> Self {
141        self.wasm_features = wasm_features;
142        self
143    }
144
145    #[wasm_bindgen]
146    #[must_use]
147    pub fn wasm_features(&self, flag: &str) -> Option<Vec<WasmFeature>> {
148        self.wasm_features.get(flag).cloned()
149    }
150
151    #[wasm_bindgen]
152    pub fn to_js(&self) -> HQResult<JsValue> {
153        serde_wasm_bindgen::to_value(&self)
154            .map_err(|_| make_hq_bug!("couldn't convert FlagInfo to JsValue"))
155    }
156}

158macro_rules! stringmap {
159    ($($k:ident : $v:expr),+ $(,)?) => {{
160        BTreeMap::from([$((String::from(stringify!($k)), $v),)+])
161    }}
162}

164/// stringifies the name of a type whilst ensuring that the type is valid
165macro_rules! ty_str {
166    ($ty:ty) => {{
167        let _ = core::any::TypeId::of::<$ty>(); // forces the type to be valid
168        stringify!($ty)
169    }};

194impl WasmFlags {

213    #[wasm_bindgen(constructor)]
214    #[must_use]
215    pub fn new(wasm_features: Vec<WasmFeature>) -> Self {
216        Self {
217            wasm_opt: Switch::On,
218            string_type: if wasm_features.contains(&WasmFeature::JSStringBuiltins) {
219                WasmStringType::JsStringBuiltins
220            } else {
221                WasmStringType::ExternRef
222            },
223            print_ir: Switch::Off,
224            integers: Switch::Off,
225            list_type: if wasm_features.contains(&WasmFeature::GC) {
226                ListType::GCArray
227            } else {
228                ListType::LinearMemory
229            },
230            unroll_loops: 2,
231            var_type_convergence: VarTypeConvergence::Tight,
232            do_ssa: Switch::On,
233            eager_number_parsing: Switch::On,
234            variable_merging: Switch::On,
235        }
236    }

238    #[wasm_bindgen]
239    #[must_use]
240    pub fn flag_info(flag: &str) -> FlagInfo {
241        match flag {
242            "string_type" => FlagInfo::new()
243                .with_name("Internal string representation")
244                .with_description(
245                    "ExternRef - uses JavaScript strings.\
246                    <br>\
247                    JsStringBuiltins (recommended) - uses JavaScript strings with the JS String Builtins proposal",
248                )
249                .with_ty(ty_str!(WasmStringType))
250                .with_wasm_features(stringmap! {
251                    ExternRef : vec![WasmFeature::ReferenceTypes],
252                    JsStringBuiltins : vec![WasmFeature::ReferenceTypes, WasmFeature::JSStringBuiltins],
253                }),
254            "list_type" => FlagInfo::new()
255                .with_name("List representation")
256                .with_description(
257                    "GCArray (recommended) - uses GC arrays.\
258                    <br>\
259                    LinearMemory (unimplemented) - stores lists in linear memory."
260                )
261                .with_ty(ty_str!(ListType))
262                .with_wasm_features(stringmap! {
263                    GCArray : vec![WasmFeature::GC],
264                    LinearMemory : vec![]
265                }),
266            // "memory_layout" => FlagInfo::new()
267            //     .with_name("Memory layout")
268            //     .with_description(
269            //         "Contiguous - stores everything in one contiguous block of memory.\
270            //         <br>\
271            //         MultiMemory (recommended) - uses multiple memories to avoid reallocations."
272            //     )
273            //     .with_ty(ty_str!(MemoryLayout))
274            //     .with_wasm_features(stringmap! {
275            //         Contiguous : vec![],
276            //         LinearMemory : vec![WasmFeature::MultiMemory]
277            //     }),
278            "wasm_opt" => FlagInfo::new()
279                .with_name("WASM optimisation")
280                .with_description("Should we try to optimise generated WASM modules using wasm-opt?")
281                .with_ty(ty_str!(Switch)),
282            "print_ir" => FlagInfo::new()
283                .with_name("Print IR")
284                .with_description("For debugging purposes only")
285                .with_ty(ty_str!(Switch)),
286            "integers" => FlagInfo::new()
287                .with_name("Integers")
288                .with_description("Emit integer instructions wherever possible. May make things faster at \
289                the cost of possible overflow, or may slow things down if mixed with floats.")
290                .with_ty(ty_str!(Switch)),
291            "unroll_loops" => FlagInfo::new()
292                .with_name("Unroll loops")
293                .with_description("Unroll the first n iterations of loops. Can help with type analysis.\
294                <br>\
295                Recommended: 2; set to 0 to disable loop unrolling.")
296                .with_ty(ty_str!(u32)),
297            "var_type_convergence" => FlagInfo::new()
298                .with_name("Variable type convergence")
299                .with_description("How much work to spend on figuring out tight bounds on variable types, \
300                vs how much performance we get out of that.\
301                <br>\
302                Any - don't try to tighten type bounds at all, giving better compilation speed at the \
303                cost of (possibly significantly) worse performance. This implies that SSA is disabled.\
304                <br>\
305                Base - tighten to the nearest base type(s) - longer compilation for good runtime performance\
306                <br>\
307                Tight - find the tightest type bounds, giving slightly worse compilation time for slightly \
308                better runtime performance")
309                .with_ty(ty_str!(VarTypeConvergence)),
310            "do_ssa" => FlagInfo::new()
311                .with_name("SSA")
312                .with_description("Should we produce an SSA form?\
313                <br>\
314                This can improve runtime performance but can drastically slow down WASM optimisation, \
315                or produce modules too large for the browser to compile.")
316                .with_ty(ty_str!(Switch)),
317            "eager_number_parsing" => FlagInfo::new()
318                .with_name("Eagerly parse strings as numbers")
319                .with_description("Eagerly convert strings to numbers at compile time.\
320                <br>\
321                Improves performance but breaks scratch compatibility for list_contents; \
322                this behaviour matches TurboWarp.")
323                .with_ty(ty_str!(Switch)),
324            "variable_merging" => FlagInfo::new()
325                .with_name("Merge variables")
326                .with_description("Merges variables of the same type. Can improve wasm-opt performance.")
327                .with_ty(ty_str!(Switch)),
328            _ => FlagInfo::new().with_name(format!("unknown setting '{flag}'").as_str()),
329        }
330    }

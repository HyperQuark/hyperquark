28pub type GlobalRegistry =
29    MapRegistry<Box<str>, (ValType, ConstExpr, GlobalMutable, GlobalExportable)>;
30
31impl GlobalRegistry {
32    pub fn finish(
33        self,
34        globals: &mut GlobalSection,
35        exports: &mut ExportSection,
36        imported_global_count: u32,
37        _imported_function_count: u32,
38        _static_function_count: u32,
39    ) {
40        for (key, (ty, suggested_initial, mutable, export)) in self.registry().take() {
41            if *export {
42                exports.export(
43                    &key,
44                    ExportKind::Global,
45                    imported_global_count + globals.len(),
46                );
47            }
48            globals.global(
49                GlobalType {
50                    val_type: ty,
51                    mutable: *mutable,
52                    shared: false,
53                },
54                &suggested_initial,
55            );
56        }
57    }
58}

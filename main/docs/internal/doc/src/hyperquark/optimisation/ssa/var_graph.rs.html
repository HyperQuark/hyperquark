16#[derive(Clone, Debug)]
17pub enum StackOperation {
18    Drop,
19    Push(VarTarget),
20    Pop(VarTarget),
21    /// this shouldn't be anything that branches or mutates variables in any way,
22    /// i.e. not loop, if/else, yield, set variable, etc
23    Opcode(IrOpcode),
24}

26#[derive(Clone, Debug, PartialOrd, PartialEq, Ord, Eq, Hash)]
27pub enum VarTarget {
28    Var(RcVar),
29    List(RcList),
30}

32impl VarTarget {
33    pub fn possible_types(&self) -> core::cell::Ref<'_, IrType> {
34        match self {
35            Self::Var(var) => var.possible_types(),
36            Self::List(list) => list.possible_types(),
37        }
38    }
39
40    pub fn add_type(&self, ty: IrType) {
41        match self {
42            Self::Var(var) => var.add_type(ty),
43            Self::List(list) => list.add_type(ty),
44        }
45    }
46}

54#[derive(Clone)]
55pub struct VariableMaps<'a> {
56    /// global variable -> local variable
57    pub ssa: BTreeMap<RcVar, RcVar>,
58    /// global variable -> (arg index, arg variable)
59    pub proc_args: &'a BTreeMap<RcVar, (usize, RcVar)>,
60}
61
62impl<'a> VariableMaps<'a> {
63    pub const fn new_with_proc_args(proc_args: &'a BTreeMap<RcVar, (usize, RcVar)>) -> Self {
64        Self {
65            ssa: BTreeMap::new(),
66            proc_args,
67        }
68    }
69
70    fn use_current_ssa_for_global<S, A, G, T>(
71        &self,
72        global_var: &RcVar,
73        if_ssa: S,
74        if_arg: A,
75        if_global: G,
76    ) -> T
77    where
78        S: FnOnce(&RcVar) -> T,
79        A: FnOnce(usize, &RcVar) -> T,
80        G: FnOnce() -> T,
81    {
82        self.ssa.get(global_var).map_or_else(
83            || {
84                if let Some((arg_index, arg_var)) = self.proc_args.get(global_var) {
85                    if_arg(*arg_index, arg_var)
86                } else {
87                    if_global()
88                }
89            },
90            if_ssa,
91        )
92    }
93}

95/// Is this edge a backlink or not?

97/// We could calculate this dynamically during the reduction stage but seeing as we can know this
98/// when constructing the graph, we might as well do it now.

100pub enum EdgeType {
101    Forward,
102    BackLink,
103}

122impl VarGraph {
123    pub fn new() -> Self {
124        let mut graph = StableDiGraph::new();
125        let first_node = graph.add_node(None);
126        let exit_node = graph.add_node(None);
127        graph.add_edge(first_node, exit_node, EdgeType::Forward);
128        Self(Rc::new(BaseVarGraph {
129            graph: RefCell::new(graph),
130            exit_node: RefCell::new(exit_node),
131        }))
132    }
133
134    pub fn graph(&self) -> &RefCell<StableDiGraph<NodeWeight, EdgeType>> {
135        &self.0.graph
136    }
137
138    fn exit_node(&self) -> &RefCell<NodeIndex> {
139        &self.0.exit_node
140    }
141
142    fn add_node(&self, weight: NodeWeight) -> NodeIndex {
143        self.graph().borrow_mut().add_node(weight)
144    }
145
146    fn add_node_at_end(&self, weight: NodeWeight) {
147        let new_node = self.add_node(weight);
148        let end_node = *self.exit_node().borrow();
149        self.add_edge(end_node, new_node, EdgeType::Forward);
150        *self.exit_node().borrow_mut() = new_node;
151    }
152
153    fn add_edge(&self, a: NodeIndex, b: NodeIndex, weight: EdgeType) -> EdgeIndex {
154        self.graph().borrow_mut().add_edge(a, b, weight)
155    }

157    /// Visits a step to split its variables and construct a variable graph for it.
158    pub fn visit_step<S>(
159        &mut self,
160        step: S,
161        variable_maps: &mut VariableMaps<'_>,
162        graphs: &mut BTreeMap<Box<str>, MaybeGraph>,
163        next_steps: &mut Vec<StepIndex>,
164        do_ssa: bool,
165    ) -> HQResult<()>
166    where
167        S: Deref<Target = RefCell<Step>>,
168    {
169        if let Some(MaybeGraph::Inlined | MaybeGraph::Finished(_)) =
170            graphs.get(step.try_borrow()?.id())
171        {
172            // we've already visited this step.
173            return Ok(());
174        }
175
176        let maybe_proc_context = {
177            let step_tmp = step.try_borrow()?;
178            step_tmp.context().proc_context.clone()
179        };
180
181        let mut should_propagate_ssa =
182            step.try_borrow()?.used_non_inline() && maybe_proc_context.is_none();
183        let mut step_ended_on_stop = false;
184
185        let mut opcode_replacements: Vec<(usize, IrOpcode)> = vec![];
186        let mut additional_opcodes: Vec<(usize, Vec<IrOpcode>)> = vec![];
187        'opcode_loop: for (i, opcode) in step.try_borrow()?.opcodes().iter().enumerate() {
188            // let's just assume that input types match up.
189            #[expect(clippy::wildcard_enum_match_arm, reason = "too many variants to match")]
190            match opcode {
191                IrOpcode::data_setvariableto(DataSetvariabletoFields {
192                    var,
193                    local_write: locality,
194                    first_write,
195                }) => {
196                    let already_local = *locality.try_borrow()?;
197                    if already_local {
198                        if var.try_borrow()?.possible_types().is_none() {
199                            let pop_node = self.add_node(Some(StackOperation::Pop(
200                                VarTarget::Var(var.try_borrow()?.clone()),
201                            )));
202                            let last_node = *self.exit_node().borrow();
203                            self.add_edge(last_node, pop_node, EdgeType::Forward);
204                            *self.exit_node().borrow_mut() = pop_node;
205                        } else {
206                            let drop_node = self.add_node(Some(StackOperation::Drop));
207                            let last_node = *self.exit_node().borrow();
208                            self.add_edge(last_node, drop_node, EdgeType::Forward);
209                            *self.exit_node().borrow_mut() = drop_node;
210                        }
211                        continue 'opcode_loop;
212                    }
213                    if !do_ssa {
214                        let pop_node = self.add_node(Some(StackOperation::Pop(VarTarget::Var(
215                            var.try_borrow()?.clone(),
216                        ))));
217                        let last_node = *self.exit_node().borrow();
218                        self.add_edge(last_node, pop_node, EdgeType::Forward);
219                        *self.exit_node().borrow_mut() = pop_node;
220                        continue 'opcode_loop;
221                    }
222                    let new_variable = RcVar::new_empty();
223                    {
224                        variable_maps
225                            .ssa
226                            .insert(var.try_borrow()?.clone(), new_variable.clone());
227                    }
228                    *var.try_borrow_mut()? = new_variable;
229                    *first_write.try_borrow_mut()? = true;
230                    *locality.try_borrow_mut()? = true;
231                    let pop_node = self.add_node(Some(StackOperation::Pop(VarTarget::Var(
232                        var.try_borrow()?.clone(),
233                    ))));
234                    let last_node = *self.exit_node().borrow();
235                    self.add_edge(last_node, pop_node, EdgeType::Forward);
236                    *self.exit_node().borrow_mut() = pop_node;
237                }
238                IrOpcode::data_teevariable(DataTeevariableFields {
239                    var,
240                    local_read_write: locality,
241                }) => {
242                    let already_local = *locality.try_borrow()?;
243                    if !do_ssa || already_local {
244                        self.add_node_at_end(Some(StackOperation::Push(VarTarget::Var(
245                            var.try_borrow()?.clone(),
246                        ))));
247                        continue 'opcode_loop;
248                    }
249                    let new_variable = RcVar::new_empty();
250                    {
251                        variable_maps
252                            .ssa
253                            .insert(var.try_borrow()?.clone(), new_variable.clone());
254                    }
255                    *var.try_borrow_mut()? = new_variable;
256                    *locality.try_borrow_mut()? = true;
257                    self.add_node_at_end(Some(StackOperation::Pop(VarTarget::Var(
258                        var.try_borrow()?.clone(),
259                    ))));
260                    self.add_node_at_end(Some(StackOperation::Push(VarTarget::Var(
261                        var.try_borrow()?.clone(),
262                    ))));
263                }
264                IrOpcode::data_variable(DataVariableFields { var, local_read }) => {
265                    if *local_read.try_borrow()? {
266                        self.add_node_at_end(Some(StackOperation::Push(VarTarget::Var(
267                            var.try_borrow()?.clone(),
268                        ))));
269                        continue;
270                    }
271                    let gvar = &var.try_borrow()?.clone();
272                    let mut var_mut = var.try_borrow_mut()?;
273                    self.add_node_at_end(Some(StackOperation::Push(VarTarget::Var(
274                        variable_maps.use_current_ssa_for_global::<_, _, _, HQResult<_>>(
275                            gvar,
276                            |current_ssa_var| {
277                                *var_mut = current_ssa_var.clone();
278                                *local_read.try_borrow_mut()? = true;
279                                Ok(current_ssa_var.clone())
280                            },
281                            |var_index, arg_var| {
282                                opcode_replacements.push((
283                                    i,
284                                    IrOpcode::procedures_argument(ProceduresArgumentFields {
285                                        index: var_index,
286                                        arg_var: arg_var.clone(),
287                                        in_warped: true,
288                                        arg_vars: Rc::clone(
289                                            &maybe_proc_context
290                                                .as_ref()
291                                                .ok_or_else(|| {
292                                                    make_hq_bug!(
293                                                        "tried to access proc argument when proc \
294                                                         context was None"
295                                                    )
296                                                })?
297                                                .arg_vars,
298                                        ),
299                                    }),
300                                ));
301                                Ok(arg_var.clone())
302                            },
303                            || Ok(gvar.clone()),
304                        )?,
305                    ))));
306                }
307                IrOpcode::data_itemoflist(DataItemoflistFields { list }) => {
308                    self.add_node_at_end(Some(StackOperation::Drop));
309                    self.add_node_at_end(Some(StackOperation::Push(VarTarget::List(list.clone()))));
310                }
311                IrOpcode::data_addtolist(DataAddtolistFields { list }) => {
312                    self.add_node_at_end(Some(StackOperation::Pop(VarTarget::List(list.clone()))));
313                }
314                IrOpcode::data_replaceitemoflist(DataReplaceitemoflistFields { list })
315                | IrOpcode::data_insertatlist(DataInsertatlistFields { list }) => {
316                    self.add_node_at_end(Some(StackOperation::Pop(VarTarget::List(list.clone()))));
317                    self.add_node_at_end(Some(StackOperation::Drop));
318                }
319                IrOpcode::control_if_else(ControlIfElseFields {
320                    branch_if,
321                    branch_else,
322                }) => {
323                    // the top item on the stack should be consumed by control_if_else
324                    self.add_node_at_end(Some(StackOperation::Drop));
325                    let last_node = *self.exit_node().borrow();
326
327                    let branch_if_mut = Rc::new(Rc::unwrap_or_clone(Rc::clone(branch_if)));
328                    let branch_else_mut = Rc::new(Rc::unwrap_or_clone(Rc::clone(branch_else)));
329
330                    let mut branch_if_variable_maps = variable_maps.clone();
331                    graphs.insert(branch_if_mut.try_borrow()?.id().into(), MaybeGraph::Started);
332                    self.visit_step(
333                        Rc::clone(&branch_if_mut),
334                        &mut branch_if_variable_maps,
335                        graphs,
336                        next_steps,
337                        do_ssa,
338                    )?;
339                    graphs.insert(branch_if_mut.try_borrow()?.id().into(), MaybeGraph::Inlined);
340                    let mut if_branch_exit = *self.exit_node().borrow();
341                    *self.exit_node().borrow_mut() = last_node;
342
343                    let mut branch_else_variable_maps = variable_maps.clone();
344                    graphs.insert(
345                        branch_else_mut.try_borrow()?.id().into(),
346                        MaybeGraph::Started,
347                    );
348                    self.visit_step(
349                        Rc::clone(&branch_else_mut),
350                        &mut branch_else_variable_maps,
351                        graphs,
352                        next_steps,
353                        do_ssa,
354                    )?;
355                    graphs.insert(
356                        branch_else_mut.try_borrow()?.id().into(),
357                        MaybeGraph::Inlined,
358                    );
359                    let mut else_branch_exit = *self.exit_node().borrow();
360                    *self.exit_node().borrow_mut() = self.add_node(None);
361
362                    if do_ssa {
363                        self.ssa_phi(
364                            vec![
365                                (
366                                    &branch_if_mut,
367                                    branch_if_variable_maps.ssa,
368                                    &mut if_branch_exit,
369                                ),
370                                (
371                                    &branch_else_mut,
372                                    branch_else_variable_maps.ssa,
373                                    &mut else_branch_exit,
374                                ),
375                            ],
376                            variable_maps,
377                            &BTreeMap::new(),
378                        )?;
379                    }
380
381                    opcode_replacements.push((
382                        i,
383                        IrOpcode::control_if_else(ControlIfElseFields {
384                            branch_if: branch_if_mut,
385                            branch_else: branch_else_mut,
386                        }),
387                    ));
388
389                    let exit_node = *self.exit_node().borrow();
390                    self.add_edge(if_branch_exit, exit_node, EdgeType::Forward);
391                    self.add_edge(else_branch_exit, exit_node, EdgeType::Forward);
392                }
393                IrOpcode::control_loop(fields) => {
394                    self.loop_ssa(
395                        &step,
396                        variable_maps,
397                        graphs,
398                        next_steps,
399                        &mut opcode_replacements,
400                        &mut additional_opcodes,
401                        maybe_proc_context.as_ref(),
402                        i,
403                        fields,
404                        do_ssa,
405                    )?;
406                }
407                IrOpcode::procedures_argument(ProceduresArgumentFields { arg_var, .. }) => {
408                    self.add_node_at_end(Some(StackOperation::Push(VarTarget::Var(
409                        arg_var.clone(),
410                    ))));
411                }
412                IrOpcode::procedures_call_warp(ProceduresCallWarpFields { proc }) => {
413                    let Some(warped_specific_proc) = &*proc.warped_specific_proc() else {
414                        hq_bug!("tried to call_warp with no warped proc")
415                    };
416                    for arg_var in warped_specific_proc
417                        .arg_vars()
418                        .try_borrow()?
419                        .iter()
420                        .rev()
421                        .cloned()
422                    {
423                        let pop_node =
424                            self.add_node(Some(StackOperation::Pop(VarTarget::Var(arg_var))));
425                        let last_node = *self.exit_node().borrow();
426                        self.add_edge(last_node, pop_node, EdgeType::Forward);
427                        *self.exit_node().borrow_mut() = pop_node;
428                    }
429                    for ret_var in warped_specific_proc
430                        .return_vars()
431                        .try_borrow()?
432                        .iter()
433                        .cloned()
434                    {
435                        let push_node =
436                            self.add_node(Some(StackOperation::Push(VarTarget::Var(ret_var))));
437                        let last_node = *self.exit_node().borrow();
438                        self.add_edge(last_node, push_node, EdgeType::Forward);
439                        *self.exit_node().borrow_mut() = push_node;
440                    }
441                }
442                IrOpcode::procedures_call_nonwarp(ProceduresCallNonwarpFields {
443                    proc,
444                    next_step,
445                }) => {
446                    let Some(nonwarped_specific_proc) = &*proc.nonwarped_specific_proc() else {
447                        hq_bug!("tried to call_nonwarp with no non-warped proc")
448                    };
449                    for arg_var in nonwarped_specific_proc
450                        .arg_vars()
451                        .try_borrow()?
452                        .iter()
453                        .rev()
454                        .cloned()
455                    {
456                        let pop_node =
457                            self.add_node(Some(StackOperation::Pop(VarTarget::Var(arg_var))));
458                        let last_node = *self.exit_node().borrow();
459                        self.add_edge(last_node, pop_node, EdgeType::Forward);
460                        *self.exit_node().borrow_mut() = pop_node;
461                    }
462                    next_steps.push(*next_step);
463                    should_propagate_ssa = true;
464                    break 'opcode_loop;
465                }
466                IrOpcode::hq_yield(HqYieldFields { mode }) => match mode {
467                    YieldMode::Inline(inline_step) => {
468                        let step_mut = Rc::new(Rc::unwrap_or_clone(Rc::clone(inline_step)));
469                        graphs.insert(step_mut.try_borrow()?.id().into(), MaybeGraph::Started);
470                        self.visit_step(
471                            Rc::clone(&step_mut),
472                            variable_maps,
473                            graphs,
474                            next_steps,
475                            do_ssa,
476                        )?;
477                        graphs.insert(step_mut.try_borrow()?.id().into(), MaybeGraph::Inlined);
478                    }
479                    YieldMode::None | YieldMode::Return => {
480                        should_propagate_ssa = true;
481                        step_ended_on_stop = true;
482                        break 'opcode_loop;
483                    }
484                    YieldMode::Schedule(_) => (), // handled at bottom of loop
485                },
486                _ => {
487                    let opcode_node = self.add_node(Some(StackOperation::Opcode(opcode.clone())));
488                    let last_node = *self.exit_node().borrow();
489                    self.add_edge(last_node, opcode_node, EdgeType::Forward);
490                    *self.exit_node().borrow_mut() = opcode_node;
491                }
492            }
493            if let Some(next_step) = opcode.yields_to_next_step() {
494                next_steps.push(next_step);
495                should_propagate_ssa = true;
496                break 'opcode_loop;
497            }
498        }
499
500        if (step.try_borrow()?.used_non_inline() || step_ended_on_stop)
501            && step.try_borrow()?.context().warp
502            && let Some(proc_context) = maybe_proc_context.as_ref()
503        {
504            for ret_var in (*proc_context.ret_vars).borrow().iter().rev().cloned() {
505                let pop_node = self.add_node(Some(StackOperation::Pop(VarTarget::Var(ret_var))));
506                let last_node = *self.exit_node().borrow();
507                self.add_edge(last_node, pop_node, EdgeType::Forward);
508                *self.exit_node().borrow_mut() = pop_node;
509            }
510        }
511
512        // extra block so the `step_mut` mutable borrow gets dropped before
513        // `propagate_ssa` is called
514        {
515            let mut step_mut = step.try_borrow_mut()?;
516            let opcodes = step_mut.opcodes_mut();
517
518            // enact opcode replacements
519            for (index, opcode_replacement) in opcode_replacements {
520                *opcodes
521                    .get_mut(index)
522                    .ok_or_else(|| make_hq_bug!("opcode index out of bounds"))? =
523                    opcode_replacement;
524            }
525
526            // insert additional opcodes
527            for (index, these_additional_opcodes) in additional_opcodes.into_iter().rev() {
528                opcodes.splice(index..index, these_additional_opcodes);
529            }
530        }
531
532        if should_propagate_ssa {
533            self.propagate_ssa(&step, variable_maps)?;
534        }
535        Ok(())
536    }

538    fn make_loop_prelude(
539        &self,
540        variable_maps: &mut VariableMaps,
541        maybe_proc_context: Option<&ProcContext>,
542        new_var_map: &BTreeMap<RcVar, RcVar>,
543    ) -> HQResult<Vec<IrOpcode>> {
544        Ok(new_var_map
545            .iter()
546            .map(|(global_var, new_var)| {
547                let (var_access, accessed_var) = variable_maps
548                    .use_current_ssa_for_global::<_, _, _, HQResult<_>>(
549                        global_var,
550                        |ssa_var| {
551                            Ok((
552                                IrOpcode::data_variable(DataVariableFields {
553                                    var: RefCell::new(ssa_var.clone()),
554                                    local_read: RefCell::new(true),
555                                }),
556                                ssa_var.clone(),
557                            ))
558                        },
559                        |arg_index, arg_var| {
560                            Ok((
561                                IrOpcode::procedures_argument(ProceduresArgumentFields {
562                                    index: arg_index,
563                                    arg_var: arg_var.clone(),
564                                    in_warped: true,
565                                    arg_vars: Rc::clone(
566                                        &maybe_proc_context
567                                            .as_ref()
568                                            .ok_or_else(|| {
569                                                make_hq_bug!(
570                                                    "tried to access proc argument when proc \
571                                                     context was None"
572                                                )
573                                            })?
574                                            .arg_vars,
575                                    ),
576                                }),
577                                arg_var.clone(),
578                            ))
579                        },
580                        || {
581                            Ok((
582                                IrOpcode::data_variable(DataVariableFields {
583                                    var: RefCell::new(global_var.clone()),
584                                    local_read: RefCell::new(false),
585                                }),
586                                global_var.clone(),
587                            ))
588                        },
589                    )?;
590                let push_node =
591                    self.add_node(Some(StackOperation::Push(VarTarget::Var(accessed_var))));
592                let pop_node =
593                    self.add_node(Some(StackOperation::Pop(VarTarget::Var(new_var.clone()))));
594                let exit_node = *self.exit_node().borrow();
595                self.add_edge(exit_node, push_node, EdgeType::Forward);
596                self.add_edge(push_node, pop_node, EdgeType::Forward);
597                *self.exit_node().borrow_mut() = pop_node;
598                variable_maps
599                    .ssa
600                    .insert(global_var.clone(), new_var.clone());
601                Ok([
602                    var_access,
603                    IrOpcode::data_setvariableto(DataSetvariabletoFields {
604                        var: RefCell::new(new_var.clone()),
605                        local_write: RefCell::new(true),
606                        first_write: RefCell::new(true),
607                    }),
608                ])
609            })
610            .collect::<HQResult<Vec<_>>>()?
611            .into_iter()
612            .flatten()
613            .collect())
614    }

616    fn loop_ssa<S>(
617        &mut self,
618        step: &S,
619        variable_maps: &mut VariableMaps,
620        graphs: &mut BTreeMap<Box<str>, MaybeGraph>,
621        next_steps: &mut Vec<StepIndex>,
622        opcode_replacements: &mut Vec<(usize, IrOpcode)>,
623        additional_opcodes: &mut Vec<(usize, Vec<IrOpcode>)>,
624        maybe_proc_context: Option<&ProcContext>,
625        i: usize,
626        ControlLoopFields {
627            first_condition,
628            condition,
629            body,
630            flip_if,
631            pre_body,
632        }: &ControlLoopFields,
633        do_ssa: bool,
634    ) -> HQResult<()>
635    where
636        S: Deref<Target = RefCell<Step>>,
637    {
638        let new_var_map: BTreeMap<_, _> = step
639            .try_borrow()?
640            .globally_scoped_variables()?
641            .map(|global_var| (global_var, RcVar::new_empty()))
642            .collect();
643        if do_ssa {
644            additional_opcodes.push((
645                i,
646                self.make_loop_prelude(variable_maps, maybe_proc_context, &new_var_map)?,
647            ));
648        }
649        if let Some(first_condition_step) = first_condition {
650            let first_condition_mut = Rc::new(Rc::unwrap_or_clone(Rc::clone(first_condition_step)));
651            graphs.insert(
652                first_condition_mut.try_borrow()?.id().into(),
653                MaybeGraph::Started,
654            );
655            self.visit_step(
656                Rc::clone(&first_condition_mut),
657                // we don't need to keep track of this map as no variables should be set in the first condition
658                &mut variable_maps.clone(),
659                graphs,
660                next_steps,
661                do_ssa,
662            )?;
663            graphs.insert(
664                first_condition_mut.try_borrow()?.id().into(),
665                MaybeGraph::Inlined,
666            );
667            let fst_drop_node = self.add_node(Some(StackOperation::Drop)); // we consume the top item on the type stack as an i32.eqz
668            let fst_end_node = *self.exit_node().borrow();
669            self.add_edge(fst_end_node, fst_drop_node, EdgeType::Forward);
670            *self.exit_node().borrow_mut() = fst_drop_node;
671            let first_cond_exit = fst_drop_node;
672
673            let header_node = self.add_node(None);
674            self.add_edge(first_cond_exit, header_node, EdgeType::Forward);
675            *self.exit_node().borrow_mut() = header_node;
676
677            let pre_body_mut = pre_body
678                .as_ref()
679                .map(|pre_body_real| -> HQResult<_> {
680                    let pre_body_mut = Rc::new(Rc::unwrap_or_clone(Rc::clone(pre_body_real)));
681                    let mut pre_body_variable_maps = variable_maps.clone();
682                    graphs.insert(pre_body_mut.try_borrow()?.id().into(), MaybeGraph::Started);
683                    self.visit_step(
684                        Rc::clone(&pre_body_mut),
685                        &mut pre_body_variable_maps,
686                        graphs,
687                        next_steps,
688                        do_ssa,
689                    )?;
690                    graphs.insert(pre_body_mut.try_borrow()?.id().into(), MaybeGraph::Inlined);
691                    Ok(pre_body_mut)
692                })
693                .transpose()?;
694
695            let body_mut = Rc::new(Rc::unwrap_or_clone(Rc::clone(body)));
696            let mut body_variable_maps = variable_maps.clone();
697            graphs.insert(body_mut.try_borrow()?.id().into(), MaybeGraph::Started);
698            self.visit_step(
699                Rc::clone(&body_mut),
700                &mut body_variable_maps,
701                graphs,
702                next_steps,
703                do_ssa,
704            )?;
705            graphs.insert(body_mut.try_borrow()?.id().into(), MaybeGraph::Inlined);
706
707            let condition_mut: InlinedStep = Rc::new(Rc::unwrap_or_clone(Rc::clone(condition)));
708            graphs.insert(condition_mut.try_borrow()?.id().into(), MaybeGraph::Started);
709            self.visit_step(
710                Rc::clone(&condition_mut),
711                &mut body_variable_maps,
712                graphs,
713                next_steps,
714                do_ssa,
715            )?;
716            graphs.insert(condition_mut.try_borrow()?.id().into(), MaybeGraph::Inlined);
717            let drop_node = self.add_node(Some(StackOperation::Drop));
718            let end_node = *self.exit_node().borrow();
719            self.add_edge(end_node, drop_node, EdgeType::Forward);
720            *self.exit_node().borrow_mut() = drop_node;
721            let mut cond_exit = drop_node;
722
723            if do_ssa {
724                self.ssa_phi(
725                    vec![(&condition_mut, body_variable_maps.ssa, &mut cond_exit)],
726                    variable_maps,
727                    &new_var_map,
728                )?;
729            }
730
731            opcode_replacements.push((
732                i,
733                IrOpcode::control_loop(ControlLoopFields {
734                    body: body_mut,
735                    first_condition: Some(first_condition_mut),
736                    flip_if: *flip_if,
737                    condition: condition_mut,
738                    pre_body: pre_body_mut,
739                }),
740            ));
741
742            self.add_edge(cond_exit, header_node, EdgeType::BackLink);
743            *self.exit_node().borrow_mut() = cond_exit;
744        } else {
745            let header_node = self.add_node(None);
746            let last_node = *self.exit_node().borrow();
747            self.add_edge(last_node, header_node, EdgeType::Forward);
748            *self.exit_node().borrow_mut() = header_node;
749
750            let condition_mut = Rc::new(Rc::unwrap_or_clone(Rc::clone(condition)));
751            let mut body_variable_maps = variable_maps.clone();
752            graphs.insert(condition_mut.try_borrow()?.id().into(), MaybeGraph::Started);
753            self.visit_step(
754                Rc::clone(&condition_mut),
755                &mut body_variable_maps,
756                graphs,
757                next_steps,
758                do_ssa,
759            )?;
760            graphs.insert(condition_mut.try_borrow()?.id().into(), MaybeGraph::Inlined);
761            let drop_node = self.add_node(Some(StackOperation::Drop));
762            let end_node = *self.exit_node().borrow();
763            self.add_edge(end_node, drop_node, EdgeType::Forward);
764            *self.exit_node().borrow_mut() = drop_node;
765
766            let pre_body_mut = pre_body
767                .as_ref()
768                .map(|pre_body_real| -> HQResult<_> {
769                    let pre_body_mut = Rc::new(Rc::unwrap_or_clone(Rc::clone(pre_body_real)));
770                    let mut pre_body_variable_maps = variable_maps.clone();
771                    graphs.insert(pre_body_mut.try_borrow()?.id().into(), MaybeGraph::Started);
772                    self.visit_step(
773                        Rc::clone(&pre_body_mut),
774                        &mut pre_body_variable_maps,
775                        graphs,
776                        next_steps,
777                        do_ssa,
778                    )?;
779                    graphs.insert(pre_body_mut.try_borrow()?.id().into(), MaybeGraph::Inlined);
780                    Ok(pre_body_mut)
781                })
782                .transpose()?;
783
784            let body_mut = Rc::new(Rc::unwrap_or_clone(Rc::clone(body)));
785            graphs.insert(body_mut.try_borrow()?.id().into(), MaybeGraph::Started);
786            self.visit_step(
787                Rc::clone(&body_mut),
788                &mut body_variable_maps,
789                graphs,
790                next_steps,
791                do_ssa,
792            )?;
793            graphs.insert(body_mut.try_borrow()?.id().into(), MaybeGraph::Inlined);
794            let mut body_exit = *self.exit_node().borrow();
795
796            if do_ssa {
797                self.ssa_phi(
798                    vec![(&body_mut, body_variable_maps.ssa, &mut body_exit)],
799                    variable_maps,
800                    &new_var_map,
801                )?;
802            }
803
804            self.add_edge(body_exit, header_node, EdgeType::BackLink);
805            *self.exit_node().borrow_mut() = body_exit;
806
807            opcode_replacements.push((
808                i,
809                IrOpcode::control_loop(ControlLoopFields {
810                    flip_if: *flip_if,
811                    body: body_mut,
812                    first_condition: None,
813                    condition: condition_mut,
814                    pre_body: pre_body_mut,
815                }),
816            ));
817        }
818        let loop_exit = *self.exit_node().borrow();
819        let new_node = self.add_node(None);
820        self.add_edge(loop_exit, new_node, EdgeType::Forward);
821        *self.exit_node().borrow_mut() = new_node;
822
823        Ok(())
824    }

826    fn propagate_ssa<S>(&self, step: &S, variable_maps: &VariableMaps) -> HQResult<()>
827    where
828        S: Deref<Target = RefCell<Step>>,
829    {
830        let post_yield = {
831            if let Some(last_op) = step.try_borrow()?.opcodes().last()
832                && matches!(
833                    last_op,
834                    IrOpcode::hq_yield(_) | IrOpcode::procedures_call_nonwarp(_)
835                )
836            {
837                true
838            } else {
839                false
840            }
841        };
842        let mut step_mut = step.try_borrow_mut()?;
843        let opcodes = step_mut.opcodes_mut();
844        let yield_op = if post_yield {
845            #[expect(clippy::unwrap_used, reason = "guaranteed last element")]
846            Some(opcodes.pop().unwrap())
847        } else {
848            None
849        };
850        for (global_var, ssa_var) in &variable_maps.ssa {
851            let push_node =
852                self.add_node(Some(StackOperation::Push(VarTarget::Var(ssa_var.clone()))));
853            let pop_node = self.add_node(Some(StackOperation::Pop(VarTarget::Var(
854                global_var.clone(),
855            ))));
856            let last_node = *self.exit_node().borrow();
857            self.add_edge(last_node, push_node, EdgeType::Forward);
858            self.add_edge(push_node, pop_node, EdgeType::Forward);
859            *self.exit_node().borrow_mut() = pop_node;
860            opcodes.extend([
861                IrOpcode::data_variable(DataVariableFields {
862                    var: RefCell::new(ssa_var.clone()),
863                    local_read: RefCell::new(true),
864                }),
865                IrOpcode::data_setvariableto(DataSetvariabletoFields {
866                    var: RefCell::new(global_var.clone()),
867                    local_write: RefCell::new(false),
868                    first_write: RefCell::new(false),
869                }),
870            ]);
871        }
872        if post_yield {
873            #[expect(clippy::unwrap_used, reason = "guaranteed to be Some")]
874            opcodes.push(yield_op.unwrap());
875        }
876        Ok(())
877    }

879    /// Insert an SSA "phi function"; that is, propagate any new SSAs to the outer scope,
880    /// at a so-called dominance frontier (where different paths meet)
881    fn ssa_phi(
882        &self,
883        mut ssa_blocks: Vec<(&InlinedStep, BTreeMap<RcVar, RcVar>, &mut NodeIndex)>,
884        variable_maps: &mut VariableMaps,
885        ssa_write_map: &BTreeMap<RcVar, RcVar>,
886    ) -> HQResult<()> {
887        let mut lower_ssas: BTreeMap<_, BTreeMap<_, _>> = BTreeMap::new();
888        let write_map_was_empty = ssa_write_map.is_empty();
889
890        for (block, block_ssa, _) in &ssa_blocks {
891            for (global, block_local) in block_ssa {
892                lower_ssas
893                    .entry(global.clone())
894                    .and_modify(|ssa_block_map| {
895                        ssa_block_map.insert(Rc::clone(block), block_local.clone());
896                    })
897                    .or_insert_with(|| BTreeMap::from([(Rc::clone(block), block_local.clone())]));
898            }
899        }
900
901        let mut block_exits: BTreeMap<_, _> = ssa_blocks
902            .iter_mut()
903            .map(|(block, _block_ssas, block_exit)| (Rc::clone(block), block_exit))
904            .collect();
905
906        let mut block_var_writes = BTreeMap::new();
907
908        for (global, block_ssas) in lower_ssas {
909            let new_var = ssa_write_map
910                .get(&global)
911                .cloned()
912                .unwrap_or_else(RcVar::new_empty);
913            for block in block_exits.keys() {
914                if !block_ssas.contains_key(block) {
915                    block_var_writes
916                        .entry(Rc::clone(block))
917                        .or_insert_with(Vec::new)
918                        .push((
919                            variable_maps
920                                .ssa
921                                .get(&global)
922                                .cloned()
923                                .map_or_else(|| (global.clone(), false), |var| (var, true)),
924                            new_var.clone(),
925                        ));
926                }
927            }
928            for (block, block_ssa) in &block_ssas {
929                block_var_writes
930                    .entry(Rc::clone(block))
931                    .or_insert_with(Vec::new)
932                    .push(((block_ssa.clone(), true), new_var.clone()));
933            }
934            variable_maps.ssa.insert(global.clone(), new_var.clone());
935        }
936
937        for (block, var_writes) in &block_var_writes {
938            let Some(block_exit) = block_exits.get_mut(block) else {
939                hq_bug!("couldn't find SSA block exit")
940            };
941            let mut block_mut = block.try_borrow_mut()?;
942            let opcodes = block_mut.opcodes_mut();
943            if !matches!(
944                opcodes.last(),
945                Some(IrOpcode::hq_yield(HqYieldFields {
946                    mode: YieldMode::Return,
947                }))
948            ) {
949                opcodes.reserve_exact(var_writes.len() * 2);
950                for ((var_read, read_local), var_write) in var_writes {
951                    let push_node =
952                        self.add_node(Some(StackOperation::Push(VarTarget::Var(var_read.clone()))));
953                    let pop_node =
954                        self.add_node(Some(StackOperation::Pop(VarTarget::Var(var_write.clone()))));
955                    self.add_edge(***block_exit, push_node, EdgeType::Forward);
956                    self.add_edge(push_node, pop_node, EdgeType::Forward);
957                    ***block_exit = pop_node;
958
959                    opcodes.append(&mut vec![
960                        IrOpcode::data_variable(DataVariableFields {
961                            var: RefCell::new(var_read.clone()),
962                            local_read: RefCell::new(*read_local),
963                        }),
964                        IrOpcode::data_setvariableto(DataSetvariabletoFields {
965                            var: RefCell::new(var_write.clone()),
966                            local_write: RefCell::new(true),
967                            // TODO: might this actually be the first write?
968                            first_write: RefCell::new(write_map_was_empty),
969                        }),
970                    ]);
971                }
972            }
973        }
974
975        Ok(())
976    }

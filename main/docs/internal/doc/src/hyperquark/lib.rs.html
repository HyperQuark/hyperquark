134#[cfg(feature = "compiler")]
135#[wasm_bindgen]
136pub fn sb3_to_wasm(proj: &str, flags: wasm::WasmFlags) -> HQResult<wasm::FinishedWasm> {
137    use ir::IrProject;
138    use wasm::flags::Switch;
139
140    let sb3_proj = sb3::Sb3Project::try_from(proj)?;
141    let ir_proj = IrProject::try_from_sb3(&sb3_proj, &flags)?;
142    if flags.print_ir == Switch::On {
143        crate::log("ir (before optimisation):");
144        crate::log(format!("{ir_proj}").as_str());
145    }
146    let ssa_token = optimisation::ir_optimise(&ir_proj, &flags)?;
147    if flags.print_ir == Switch::On {
148        crate::log("ir (after optimisationnnn):");
149        crate::log(format!("{ir_proj}").as_str());
150    }
151    crate::log("optimisation done");
152    wasm::WasmProject::from_ir(&ir_proj, ssa_token, flags)?.finish()
153}

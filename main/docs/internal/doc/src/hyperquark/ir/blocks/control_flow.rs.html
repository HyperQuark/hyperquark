13pub fn generate_loop(
14    warp: bool,
15    should_break: &mut bool,
16    block_info: &BlockInfo,
17    blocks: &BTreeMap<Box<str>, Block>,
18    context: &StepContext,
19    final_next_blocks: NextBlocks,
20    first_condition_instructions: Option<Vec<IrOpcode>>,
21    condition_instructions: Vec<IrOpcode>,
22    pre_body_instructions: Option<Vec<IrOpcode>>,
23    flip_if: bool,
24    setup_instructions: Vec<IrOpcode>,
25    flags: &WasmFlags,
26) -> HQResult<Vec<IrOpcode>> {
27    let substack_id = match block_info.inputs.get("SUBSTACK") {
28        Some(
29            Input::NoShadow(_, Some(substack_input)) | Input::Shadow(_, Some(substack_input), _),
30        ) => {
31            let BlockArrayOrId::Id(id) = substack_input else {
32                hq_bad_proj!("malformed SUBSTACK input")
33            };
34            Some(id)
35        }
36        _ => None,
37    };
38
39    let substack_block = if let Some(id) = substack_id {
40        Some(
41            blocks
42                .get(id)
43                .ok_or_else(|| make_hq_bad_proj!("SUBSTACK block doesn't seem to exist"))?,
44        )
45    } else {
46        None
47    };
48    if warp {
49        // TODO: can this be expressed in the same way as non-warping loops,
50        // just with yield_first: false?
51        let substack_blocks = if let Some(block) = substack_block {
52            from_block(
53                block,
54                blocks,
55                context,
56                &context.target().project(),
57                NextBlocks::new(false),
58                flags,
59            )?
60        } else {
61            vec![]
62        };
63        let substack_step = Rc::new(RefCell::new(Step::new(
64            None,
65            context.clone(),
66            substack_blocks,
67            context.target().project(),
68            false,
69        )));
70        let condition_step = Rc::new(RefCell::new(Step::new(
71            None,
72            context.clone(),
73            condition_instructions,
74            context.target().project(),
75            false,
76        )));
77        let first_condition_step = first_condition_instructions.map(|instrs| {
78            Rc::new(RefCell::new(Step::new(
79                None,
80                context.clone(),
81                instrs,
82                context.target().project(),
83                false,
84            )))
85        });
86        let pre_body_step = pre_body_instructions.map(|instrs| {
87            Rc::new(RefCell::new(Step::new(
88                None,
89                context.clone(),
90                instrs,
91                context.target().project(),
92                false,
93            )))
94        });
95        Ok(setup_instructions
96            .into_iter()
97            .chain(vec![IrOpcode::control_loop(ControlLoopFields {
98                first_condition: first_condition_step,
99                condition: condition_step,
100                body: substack_step,
101                pre_body: pre_body_step,
102                flip_if,
103            })])
104            .collect())
105    } else {
106        *should_break = true;
107        let next_step =
108            generate_next_step_inlined(block_info, blocks, context, final_next_blocks, flags)?;
109        let project = context.project()?;
110        let mut condition_step = Step::new(
111            None,
112            context.clone(),
113            condition_instructions.clone(),
114            context.target().project(),
115            true,
116        );
117        let substack_step = Rc::new(RefCell::new(Step::new(
118            None,
119            context.clone(),
120            vec![],
121            context.target().project(),
122            false,
123        )));
124        condition_step
125            .opcodes_mut()
126            .push(IrOpcode::control_if_else(ControlIfElseFields {
127                branch_if: Rc::clone(if flip_if { &next_step } else { &substack_step }),
128                branch_else: Rc::clone(if flip_if { &substack_step } else { &next_step }),
129            }));
130        let condition_step_index = project.new_owned_step(condition_step)?;
131        if let Some(pre_body_blocks) = pre_body_instructions {
132            substack_step
133                .try_borrow_mut()?
134                .opcodes_mut()
135                .extend(pre_body_blocks);
136        }
137        if let Some(block) = substack_block {
138            let substack_blocks = from_block(
139                block,
140                blocks,
141                context,
142                &context.target().project(),
143                NextBlocks::new(false).extend_with_inner(NextBlockInfo {
144                    yield_first: true,
145                    block: NextBlock::StepIndex(condition_step_index),
146                }),
147                flags,
148            )?;
149            substack_step
150                .try_borrow_mut()?
151                .opcodes_mut()
152                .extend(substack_blocks);
153        } else {
154            substack_step
155                .try_borrow_mut()?
156                .opcodes_mut()
157                .push(IrOpcode::hq_yield(HqYieldFields {
158                    mode: YieldMode::Schedule(condition_step_index),
159                }));
160        }
161        Ok(setup_instructions
162            .into_iter()
163            .chain(first_condition_instructions.unwrap_or(condition_instructions))
164            .chain(vec![IrOpcode::control_if_else(ControlIfElseFields {
165                branch_if: Rc::clone(if flip_if { &next_step } else { &substack_step }),
166                branch_else: Rc::clone(if flip_if { &substack_step } else { &next_step }),
167            })])
168            .collect())
169    }
170}

172#[expect(clippy::too_many_arguments, reason = "will fix later. maybe.")]
173pub fn generate_if_else(
174    if_block: (&Block, Box<str>),
175    maybe_else_block: Option<(&Block, Box<str>)>,
176    block_info: &BlockInfo,
177    final_next_blocks: &NextBlocks,
178    blocks: &BTreeMap<Box<str>, Block>,
179    context: &StepContext,
180    should_break: &mut bool,
181    flags: &WasmFlags,
182) -> HQResult<Vec<IrOpcode>> {
183    if !context.warp {
184        let this_project = context.project()?;
185        let dummy_project = Rc::new(IrProject::new(
186            this_project.global_variables().clone(),
187            this_project.global_lists().clone(),
188            Box::from(this_project.broadcasts()),
189            0,
190            vec![],
191        ));
192        let dummy_target = Rc::new(Target::new(
193            false,
194            context.target().variables().clone(),
195            context.target().lists().clone(),
196            Rc::downgrade(&dummy_project),
197            RefCell::new(context.target().procedures()?.clone()),
198            0,
199            context.target().costumes().into(),
200        ));
201        dummy_project
202            .targets()
203            .try_borrow_mut()
204            .map_err(|_| make_hq_bug!("couldn't mutably borrow cell"))?
205            .insert("".into(), Rc::clone(&dummy_target));
206        let dummy_context = StepContext {
207            target: Rc::clone(&dummy_target),
208            ..context.clone()
209        };
210        let dummy_if_step = Step::from_block(
211            if_block.0,
212            if_block.1.clone(),
213            blocks,
214            &dummy_context,
215            &Rc::downgrade(&dummy_project),
216            NextBlocks::new(false),
217            false,
218            flags,
219        )?;
220        let dummy_else_step = if let Some((else_block, else_block_id)) = maybe_else_block.clone() {
221            Step::from_block(
222                else_block,
223                else_block_id,
224                blocks,
225                &dummy_context,
226                &Rc::downgrade(&dummy_project),
227                NextBlocks::new(false),
228                false,
229                flags,
230            )?
231        } else {
232            Step::new_empty(
233                Rc::downgrade(&dummy_project),
234                false,
235                Rc::clone(&dummy_target),
236            )
237        };
238        if dummy_if_step.does_yield() || dummy_else_step.does_yield() {
239            // TODO: ideally if only one branch yields then we'd duplicate the next step and put one
240            // version inline after the branch, and the other tagged on in the substep's NextBlocks
241            // as usual, to allow for extra variable type optimisations.
242            #[expect(
243                clippy::option_if_let_else,
244                reason = "map_or_else alternative is too complex"
245            )]
246            let (next_block, next_blocks) = if let Some(ref next_block) = block_info.next {
247                (
248                    Some(NextBlock::ID(next_block.clone())),
249                    final_next_blocks.extend_with_inner(NextBlockInfo {
250                        yield_first: false,
251                        block: NextBlock::ID(next_block.clone()),
252                    }),
253                )
254            } else if let (Some(next_block_info), _) = final_next_blocks.clone().pop_inner() {
255                (Some(next_block_info.block), final_next_blocks.clone())
256            } else {
257                (
258                    None,
259                    final_next_blocks.clone(), // preserve termination behaviour
260                )
261            };
262            let final_if_step = {
263                Step::from_block(
264                    if_block.0,
265                    if_block.1,
266                    blocks,
267                    context,
268                    &context.target().project(),
269                    next_blocks.clone(),
270                    false,
271                    flags,
272                )?
273            };
274            let final_else_step = if let Some((else_block, else_block_id)) = maybe_else_block {
275                Step::from_block(
276                    else_block,
277                    else_block_id,
278                    blocks,
279                    context,
280                    &context.target().project(),
281                    next_blocks,
282                    false,
283                    flags,
284                )?
285            } else {
286                let opcode = match next_block {
287                    Some(NextBlock::ID(id)) => {
288                        let id_next_block = blocks
289                            .get(&id)
290                            .ok_or_else(|| make_hq_bad_proj!("missing next block"))?;
291                        vec![IrOpcode::hq_yield(HqYieldFields {
292                            mode: YieldMode::Inline(Rc::new(RefCell::new(Step::from_block(
293                                id_next_block,
294                                id.clone(),
295                                blocks,
296                                context,
297                                &context.target().project(),
298                                next_blocks,
299                                false,
300                                flags,
301                            )?))),
302                        })]
303                    }
304                    Some(NextBlock::Step(mut step)) => {
305                        step.make_inlined();
306                        vec![IrOpcode::hq_yield(HqYieldFields {
307                            mode: YieldMode::Inline(Rc::new(RefCell::new(step))),
308                        })]
309                    }
310                    Some(NextBlock::StepIndex(step_index)) => {
311                        let mut step = context
312                            .project()?
313                            .steps()
314                            .try_borrow()?
315                            .get(step_index.0)
316                            .ok_or_else(|| make_hq_bug!("step index out of bounds"))?
317                            .try_borrow()?
318                            .clone();
319                        step.make_inlined();
320                        vec![IrOpcode::hq_yield(HqYieldFields {
321                            mode: YieldMode::Inline(Rc::new(RefCell::new(step))),
322                        })]
323                    }
324                    None => {
325                        if next_blocks.terminating() {
326                            vec![IrOpcode::hq_yield(HqYieldFields {
327                                mode: YieldMode::None,
328                            })]
329                        } else {
330                            vec![]
331                        }
332                    }
333                };
334                Step::new(
335                    None,
336                    context.clone(),
337                    opcode,
338                    context.target().project(),
339                    false,
340                )
341            };
342            *should_break = true;
343            return Ok(vec![IrOpcode::control_if_else(ControlIfElseFields {
344                branch_if: Rc::new(RefCell::new(final_if_step)),
345                branch_else: Rc::new(RefCell::new(final_else_step)),
346            })]);
347        }
348    }
349    let final_if_step = Step::from_block(
350        if_block.0,
351        if_block.1,
352        blocks,
353        context,
354        &context.target().project(),
355        NextBlocks::new(false),
356        false,
357        flags,
358    )?;
359    let final_else_step = if let Some((else_block, else_block_id)) = maybe_else_block {
360        Step::from_block(
361            else_block,
362            else_block_id,
363            blocks,
364            context,
365            &context.target().project(),
366            NextBlocks::new(false),
367            false,
368            flags,
369        )?
370    } else {
371        Step::new(
372            None,
373            context.clone(),
374            vec![],
375            context.target().project(),
376            false,
377        )
378    };
379    Ok(vec![IrOpcode::control_if_else(ControlIfElseFields {
380        branch_if: Rc::new(RefCell::new(final_if_step)),
381        branch_else: Rc::new(RefCell::new(final_else_step)),
382    })])
383}

385pub fn generate_exhaustive_string_comparison<I, S, F>(
386    string_source: I,
387    instruction: F,
388    fallback: Vec<IrOpcode>,
389    context: &StepContext,
390    project: &Weak<IrProject>,
391    flags: &WasmFlags,
392) -> HQResult<Vec<IrOpcode>>
393where
394    I: IntoIterator<Item = S>,
395    S: Into<Box<str>> + Clone,
396    F: Fn(Box<str>) -> IrOpcode,
397{
398    let var = RcVar::new(IrType::String, &VarVal::String("".into()), None, flags)?;
399    Ok(vec![
400        IrOpcode::hq_cast(HqCastFields(IrType::String)),
401        IrOpcode::data_setvariableto(DataSetvariabletoFields {
402            var: RefCell::new(var.clone()),
403            local_write: RefCell::new(true),
404            first_write: RefCell::new(true),
405        }),
406    ]
407    .into_iter()
408    .chain(
409        string_source
410            .into_iter()
411            .fold(
412                Rc::new(RefCell::new(Step::new(
413                    None,
414                    context.clone(),
415                    fallback,
416                    Weak::clone(project),
417                    false,
418                ))),
419                |branch_else, string| {
420                    let branch_if = Rc::new(RefCell::new(Step::new(
421                        None,
422                        context.clone(),
423                        vec![instruction(string.clone().into())],
424                        Weak::clone(project),
425                        false,
426                    )));
427                    Rc::new(RefCell::new(Step::new(
428                        None,
429                        context.clone(),
430                        vec![
431                            IrOpcode::data_variable(DataVariableFields {
432                                var: RefCell::new(var.clone()),
433                                local_read: RefCell::new(true),
434                            }),
435                            IrOpcode::hq_text(HqTextFields(string.into())),
436                            IrOpcode::operator_equals, // todo: this should be a case-sensitive comparison
437                            IrOpcode::control_if_else(ControlIfElseFields {
438                                branch_if,
439                                branch_else,
440                            }),
441                        ],
442                        Weak::clone(project),
443                        false,
444                    )))
445                },
446            )
447            .try_borrow()?
448            .opcodes()
449            .clone(),
450    )
451    .collect())

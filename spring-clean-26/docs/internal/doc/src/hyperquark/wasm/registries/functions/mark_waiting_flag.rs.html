7/// Mark a waiting flag as done.
8///
9/// This is designed to be exported (as `"mark_waiting_flag"`) and called by JS.
10///
11/// Takes 1 parameter:
12/// - A nonnull struct with a single i8 field.
13///
14/// Override with one u32, the single-field i8 struct type index
15pub struct MarkWaitingFlag;
16impl NamedRegistryItem<MaybeStaticFunction> for MarkWaitingFlag {
17    const VALUE: MaybeStaticFunction = MaybeStaticFunction {
18        static_function: None,
19        maybe_populate: || None,
20    };
21}
22pub type MarkWaitingFlagOverride = u32;
23impl NamedRegistryItemOverride<MaybeStaticFunction, MarkWaitingFlagOverride> for MarkWaitingFlag {
24    fn r#override(i8_struct_ty: u32) -> MaybeStaticFunction {
25        MaybeStaticFunction {
26            static_function: Some(StaticFunction {
27                export: Some("mark_waiting_flag".into()),
28                instructions: Box::from(wasm_const![
29                    LocalGet(0),
30                    I32Const(1),
31                    StructSet {
32                        struct_type_index: i8_struct_ty,
33                        field_index: 0
34                    },
35                    End,
36                ] as &[_]),
37                params: Box::new([ValType::Ref(RefType {
38                    nullable: false,
39                    heap_type: HeapType::Concrete(i8_struct_ty),
40                })]),
41                returns: Box::new([]),
42                locals: Box::new([]),
43            }),
44            maybe_populate: || None,
45        }
46    }
47}

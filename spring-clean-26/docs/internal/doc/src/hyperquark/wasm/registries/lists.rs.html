61    pub fn array_type<N>(&self, list: &RcList) -> HQResult<N>
62    where
63        N: TryFrom<usize>,
64        <N as TryFrom<usize>>::Error: fmt::Debug,
65    {
66        let elem_type = WasmProject::ir_type_to_wasm(*list.possible_types());
67        self.types().array(StorageType::Val(elem_type), true)
68    }
69
70    pub fn register<M>(&self, list: &RcList) -> HQResult<(u32, Option<M>)>
71    where
72        M: TryFrom<usize>,
73        <M as TryFrom<usize>>::Error: fmt::Debug,
74    {
75        let array_type_index = self.array_type(list)?;
76
77        let initial_length = list
78            .initial_value()
79            .len()
80            .try_into()
81            .map_err(|_| make_hq_bug!("initial list value length out of bounds"))?;
82
83        let array_size = if *list.length_mutable().borrow() {
84            200_000
85        } else {
86            initial_length
87        };
88
89        let init_instructions = [
90            match list.possible_types().base_type() {
91                Some(IrType::Int | IrType::Boolean) => Instruction::I32Const(0),
92                Some(IrType::Float) => Instruction::F64Const(0.0.into()),
93                Some(IrType::String) => {
94                    let string_idx = self.strings().register_default("".into())?;
95                    Instruction::GlobalGet(string_idx)
96                }
97                _ => Instruction::I64Const(0),
98            },
99            Instruction::I32Const(array_size),
100            Instruction::ArrayNew(array_type_index),
101        ];
102
103        let array_global = self.globals().register(
104            format!("__rclist_list_{}", list.id()).into(),
105            (
106                ValType::Ref(RefType {
107                    nullable: false,
108                    heap_type: HeapType::Concrete(array_type_index),
109                }),
110                ConstExpr::extended(init_instructions),
111                GlobalMutable(*list.length_mutable().borrow()),
112                GlobalExportable(false),
113            ),
114        )?;
115
116        self.registry()
117            .register::<usize>(list.clone(), array_global)?;
118
119        // make sure strings are registered before we call `finish`, as `finish` needs to be
120        // called after the strings registry is finished.
121        match list.possible_types().base_type() {
122            Some(IrType::String) => {
123                list.initial_value().iter().try_for_each(|val| {
124                    let VarVal::String(s) = val else {
125                        hq_bug!("VarVal type should be included in var's possible types")
126                    };
127                    self.strings().register_default::<usize>(s.clone())?;
128                    Ok(())
129                })?;
130            }
131            None => {
132                list.initial_value()
133                    .iter()
134                    .try_for_each::<_, HQResult<()>>(|val| {
135                        if let VarVal::String(s) = val {
136                            self.tabled_strings().register_default::<usize>(s.clone())?;
137                        }
138                        Ok(())
139                    })?;
140            }
141            _ => (),
142        }
143
144        let length_global = if *list.length_mutable().borrow() {
145            Some(self.globals().register(
146                format!("__rclist_len_{}", list.id()).into(),
147                (
148                    ValType::I32,
149                    ConstExpr::i32_const(initial_length),
150                    GlobalMutable(true),
151                    GlobalExportable(false),
152                ),
153            )?)
154        } else {
155            None
156        };
157
158        Ok((array_global, length_global))
159    }

161    pub fn finish(
162        self,
163        data_section: &mut DataSection,
164        elem_section: &mut ElementSection,
165        start_func: &mut Function,
166        imported_global_count: u32,
167    ) -> HQResult<()> {
168        for (list, &array_global) in self.registry().registry().borrow().iter() {
169            start_func
170                .instruction(&Instruction::GlobalGet(
171                    array_global + imported_global_count,
172                ))
173                .instruction(&Instruction::I32Const(0))
174                .instruction(&Instruction::I32Const(0))
175                .instruction(&Instruction::I32Const(
176                    list.initial_value()
177                        .len()
178                        .try_into()
179                        .map_err(|_| make_hq_bug!("list initial value length out of bounds"))?,
180                ));
181
182            let array_type_index = self.array_type(list)?;
183
184            match list.possible_types().base_type() {
185                Some(IrType::Float) => {
186                    let floats_bytes = list
187                        .initial_value()
188                        .iter()
189                        .map(|val| {
190                            let VarVal::Float(f) = val else {
191                                hq_bug!("VarVal type should be included in var's possible types");
192                            };
193                            Ok((*f).to_le_bytes())
194                        })
195                        .collect::<HQResult<Box<[_]>>>()?
196                        .into_iter()
197                        .flatten()
198                        .collect::<Box<[_]>>();
199
200                    data_section.passive(floats_bytes);
201
202                    start_func.instruction(&Instruction::ArrayInitData {
203                        array_type_index,
204                        array_data_index: data_section.len() - 1,
205                    });
206                }
207                Some(IrType::Int) => {
208                    let ints_bytes = list
209                        .initial_value()
210                        .iter()
211                        .map(|val| {
212                            Ok(match val {
213                                VarVal::Int(i) => *i,
214                                VarVal::Bool(b) => (*b).into(),
215                                VarVal::String(_) | VarVal::Float(_) => {
216                                    hq_bug!(
217                                        "VarVal type should be included in var's possible types"
218                                    )
219                                }
220                            }
221                            .to_le_bytes())
222                        })
223                        .collect::<HQResult<Box<[_]>>>()?
224                        .into_iter()
225                        .flatten()
226                        .collect::<Box<[_]>>();
227
228                    data_section.passive(ints_bytes);
229
230                    start_func.instruction(&Instruction::ArrayInitData {
231                        array_type_index,
232                        array_data_index: data_section.len() - 1,
233                    });
234                }
235                Some(IrType::Boolean) => {
236                    // todo: use packed i8 type?
237
238                    let bools_bytes = list
239                        .initial_value()
240                        .iter()
241                        .map(|val| {
242                            let VarVal::Bool(b) = val else {
243                                hq_bug!("VarVal type should be included in var's possible types")
244                            };
245                            Ok(i32::from(*b).to_le_bytes())
246                        })
247                        .collect::<HQResult<Box<[_]>>>()?
248                        .into_iter()
249                        .flatten()
250                        .collect::<Box<[_]>>();
251
252                    data_section.passive(bools_bytes);
253
254                    start_func.instruction(&Instruction::ArrayInitData {
255                        array_type_index,
256                        array_data_index: data_section.len() - 1,
257                    });
258                }
259                Some(IrType::String) => {
260                    let strings = list
261                        .initial_value()
262                        .iter()
263                        .map(|val| {
264                            let VarVal::String(s) = val else {
265                                hq_bug!("VarVal type should be included in var's possible types")
266                            };
267                            let string_idx = self.strings().register_default(s.clone())?;
268                            Ok(ConstExpr::global_get(string_idx))
269                        })
270                        .collect::<HQResult<Box<[_]>>>()?;
271
272                    elem_section.passive(Elements::Expressions(
273                        RefType::EXTERNREF,
274                        Cow::Borrowed(&strings),
275                    ));
276
277                    start_func.instruction(&Instruction::ArrayInitElem {
278                        array_type_index,
279                        array_elem_index: elem_section.len() - 1,
280                    });
281                }
282                _ => {
283                    let boxed_bytes = list
284                        .initial_value()
285                        .iter()
286                        .map(|val| {
287                            Ok(match val {
288                                VarVal::Int(i) => i64::from(*i) | BOXED_INT_PATTERN,
289                                VarVal::Bool(b) => i64::from(*b) | BOXED_BOOL_PATTERN,
290                                VarVal::Float(f) => i64::from_le_bytes(f.to_le_bytes()),
291                                VarVal::String(s) => {
292                                    self.tabled_strings().register_default::<i64>(s.clone())?
293                                        | BOXED_STRING_PATTERN
294                                }
295                            }
296                            .to_le_bytes())
297                        })
298                        .collect::<HQResult<Box<[_]>>>()?
299                        .into_iter()
300                        .flatten()
301                        .collect::<Box<[_]>>();
302
303                    data_section.passive(boxed_bytes);
304
305                    start_func.instruction(&Instruction::ArrayInitData {
306                        array_type_index,
307                        array_data_index: data_section.len() - 1,
308                    });
309                }
310            }
311        }
312
313        Ok(())
314    }

529    pub fn output_type(
530        &self,
531        inputs: Rc<[crate::ir::IrType]>,
532    ) -> HQResult<crate::ir::ReturnType> {
533        if inputs.iter().any(crate::ir::IrType::is_none) {
534            hq_bug!("got none input type :scream: \n at opcode {:}", self)
535        }
536        if let Self::procedures_call_warp(_) = self {
537            // special case for procedure calls because they can accumulate a LOT
538            // of arguments, which makes boxed_output_type very slow, and we know
539            // that the output type does not depend upon the input types (for now).
540            return self.base_output_type(inputs);
541        }
542        boxed_output_type(|ins| self.base_output_type(ins), inputs)
543    }

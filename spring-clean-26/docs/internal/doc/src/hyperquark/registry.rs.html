70    type Value;
71}

73pub trait Registry: Sized + RegistryType {
74    fn registry(&self) -> &RefCell<IndexMap<Self::Key, Self::Value>>;
75
76    /// get the index of the specified item, inserting it if it doesn't exist in the map already.
77    /// Doesn't check if the provided value matches what's already there. This is generic because
78    /// various different numeric types are needed in different places, so it's easiest to encapsulate
79    /// the casting logic in here.
80    fn register<N>(&self, key: Self::Key, value: Self::Value) -> HQResult<N>
81    where
82        N: TryFrom<usize>,
83        <N as TryFrom<usize>>::Error: fmt::Debug,
84    {
85        self.registry()
86            .try_borrow_mut()
87            .map_err(|_| make_hq_bug!("couldn't mutably borrow cell"))?
88            .entry(key.clone())
89            .or_insert(value);
90        N::try_from(
91            self.registry()
92                .try_borrow()?
93                .get_index_of(&key)
94                .ok_or_else(|| make_hq_bug!("couldn't find entry in Registry"))?,
95        )
96        .map_err(|_| make_hq_bug!("registry item index out of bounds"))
97    }
98
99    fn register_override<N>(&self, key: Self::Key, value: Self::Value) -> HQResult<N>
100    where
101        N: TryFrom<usize>,
102        <N as TryFrom<usize>>::Error: fmt::Debug,
103    {
104        self.registry()
105            .try_borrow_mut()
106            .map_err(|_| make_hq_bug!("couldn't mutably borrow cell"))?
107            .entry(key.clone())
108            .insert_entry(value);
109        N::try_from(
110            self.registry()
111                .try_borrow()?
112                .get_index_of(&key)
113                .ok_or_else(|| make_hq_bug!("couldn't find entry in Registry"))?,
114        )
115        .map_err(|_| make_hq_bug!("registry item index out of bounds"))
116    }
117
118    // TODO: register_override_ifexists or similar - for things like mark_waiting_flag,
119    // which need to be overriden if they are registered, but don't actually need to be
120    // registered always.
121}

163/// Registry type for items that will be registered with an arbitrary "name" of some sort which is `const`able.
164///
165/// This means we can store the value statically in a trait-implementing struct, and derive
166/// the "name" from that struct (by default from the name of the struct).
167///
168/// A `NameRegistry` is defined with respect to a [`NamedRegistrar`], which can be any object
169/// (probably a ZST) that implements [`RegistryType`], defining the types
170/// [`Key`](RegistryType::Key) (defaults to [`Box<str>`]) and [`Value`](RegistryType::Value). Manually
171/// implementing [`NamedRegistrar`] is only necessary if `Key: !From<&str>` or you wish to specify a
172/// custom naming function.
173///
174/// To use a ZST `T` as an itemin a `NamedRegistry`, implement [`NamedRegistryItem<Value>`](NamedRegistryItem)
175/// for `T` to define the default value for the item.
176///
177/// If you wish items in this registry to be overridable, `impl`
178/// [`NamedRegistryItemOverride<Value, A>`](NamedRegistryItemOverride) `for T`, where `A` is an argument to
179/// pass to the [`r#override`](NamedRegistryItemOverride::override) function. If multiple arguments are
180/// needed, they should be wrapped into a tuple or struct.
181///
182/// # Examples
183/// ```
184/// # use hyperquark::prelude::*;
185/// # fn main() -> HQResult<()> {
186/// struct NumberRegistrar;
187/// impl RegistryType for NumberRegistrar {
188///     type Key = Box<str>;
189///     type Value = u32;
190/// }
191/// // `NamedRegistrar` is implemented automatically because `Key = Box<str>`
192///
193/// type NumberRegistry = NamedRegistry<NumberRegistrar>;
194///
195/// struct Zero;
196/// impl NamedRegistryItem<u32> for Zero {
197///     const VALUE: u32 = 0;
198/// }
199///
200/// struct Natural;
201/// impl NamedRegistryItem<u32> for Natural {
202///     const VALUE: u32 = 1;
203/// }
204/// impl NamedRegistryItemOverride<u32, u32> for Natural {
205///     fn r#override(new: u32) -> u32 {
206///         new
207///     }
208/// }
209///
210/// let num_reg = NumberRegistry::default();
211/// let nat_pos: usize = num_reg.register::<Natural, _>()?;
212/// let zero_pos: usize = num_reg.register::<Zero, _>()?;
213/// let also_nat_pos: usize = num_reg.register_override::<Natural, _, _>(5)?;
214/// assert_eq!(nat_pos, also_nat_pos);
215///
216/// // let also_zero_pos: usize = num_reg.register_override::<Zero, _, _>(0)?;
217/// // ^ would fail to compile because `Zero` doesn't implement `NamedRegistryItemOverride`
218///
219/// # Ok(())
220/// # }
221/// ```
222pub struct NamedRegistry<R>(MapRegistry<R::Key, R::Value>)
223where
224    R: NamedRegistrar<R>;
225
226impl<R> NamedRegistry<R>
227where
228    R: NamedRegistrar<R>,
229{
230    pub fn registry(&self) -> &RefCell<IndexMap<R::Key, R::Value>> {
231        self.0.registry()
232    }
233
234    /// Registers a `NamedRegistryItem` using its key function and its `const VALUE`
235    pub fn register<T, N>(&self) -> HQResult<N>
236    where
237        N: TryFrom<usize>,
238        <N as TryFrom<usize>>::Error: fmt::Debug,
239        T: NamedRegistryItem<R::Value>,
240    {
241        self.0.register(R::name::<T>(), T::VALUE)
242    }
243
244    /// Registers a runtime key-value pair; just calls `register` on the underlying
245    /// `Registry`
246    pub fn register_dyn<N>(&self, key: R::Key, value: R::Value) -> HQResult<N>
247    where
248        N: TryFrom<usize>,
249        <N as TryFrom<usize>>::Error: fmt::Debug,
250    {
251        self.0.register(key, value)
252    }
253
254    /// Override a runtime key-value pair with a runtime value; just calls
255    /// `register_override` on the underlying `Registry`
256    pub fn register_dyn_override<N>(&self, key: R::Key, value: R::Value) -> HQResult<N>
257    where
258        N: TryFrom<usize>,
259        <N as TryFrom<usize>>::Error: fmt::Debug,
260    {
261        self.0.register_override(key, value)
262    }
263
264    /// Overrides a `NamedRegistryItem` entry using the override argument types
265    /// associated with the corresponding `NamedRegistryItemOverride`
266    pub fn register_override<T, N, A>(&self, override_arg: A) -> HQResult<N>
267    where
268        N: TryFrom<usize>,
269        <N as TryFrom<usize>>::Error: fmt::Debug,
270        T: NamedRegistryItem<R::Value> + NamedRegistryItemOverride<R::Value, A>,
271    {
272        self.0
273            .register_override(R::name::<T>(), T::r#override(override_arg))
274    }
275
276    /// Tries to override a `NamedRegistryItem` entry using the override argument
277    /// types associated with the corresponding `TryNamedRegistryItemOverride`
278    pub fn try_register_override<T, N, A>(&self, override_arg: A) -> HQResult<N>
279    where
280        N: TryFrom<usize>,
281        <N as TryFrom<usize>>::Error: fmt::Debug,
282        T: NamedRegistryItem<R::Value> + TryNamedRegistryItemOverride<R::Value, A>,
283    {
284        self.0
285            .register_override(R::name::<T>(), T::try_override(override_arg)?)
286    }
287}

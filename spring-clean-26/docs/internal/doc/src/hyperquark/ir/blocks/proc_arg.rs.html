1use crate::instructions::{HqIntegerFields, IrOpcode, ProceduresArgumentFields};
2use crate::ir::StepContext;
3use crate::prelude::*;
4use crate::sb3::{BlockInfo, VarVal};
5
6#[derive(Copy, Clone, Debug, PartialEq, Eq)]
7pub enum ProcArgType {
8    Boolean,
9    StringNumber,
10}
11
12pub fn procedure_argument(
13    _arg_type: ProcArgType,
14    block_info: &BlockInfo,
15    context: &StepContext,
16) -> HQResult<Vec<IrOpcode>> {
17    let Some(proc_context) = context.proc_context.clone() else {
18        // this is always the default, regardless of type
19        return Ok(vec![IrOpcode::hq_integer(HqIntegerFields(0))]);
20    };
21    let VarVal::String(arg_name) = block_info
22        .fields
23        .get("VALUE")
24        .ok_or_else(|| make_hq_bad_proj!("missing VALUE field for proc argument"))?
25        .get_0()
26        .ok_or_else(|| make_hq_bad_proj!("missing value of VALUE field"))?
27    else {
28        hq_bad_proj!("non-string proc argument name")
29    };
30    let Some(index) = proc_context
31        .arg_names
32        .iter()
33        .position(|name| name == arg_name)
34    else {
35        return Ok(vec![IrOpcode::hq_integer(HqIntegerFields(0))]);
36    };
37    let arg_vars = (*proc_context.arg_vars).borrow();
38    let arg_var = arg_vars
39        .get(index)
40        .ok_or_else(|| make_hq_bad_proj!("argument index not in range of argumenttypes"))?;
41    Ok(vec![IrOpcode::procedures_argument(
42        ProceduresArgumentFields {
43            index,
44            arg_var: arg_var.clone(),
45            in_warped: context.warp,
46            arg_vars: Rc::clone(&proc_context.arg_vars),
47        },
48    )])
49}

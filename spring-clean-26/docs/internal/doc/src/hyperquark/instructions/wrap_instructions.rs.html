10//! Module for generating WASM for a sequence of instructions.
11//!
12//! Some definitions for formality:
13//!
14//! Define ``{$#`is_root` : #`Fork` -> #`bool`$}`` to be
15//!   ``{$ #`is_root`;(f) = f.#`branch`.#`is_none`;(). $}``
16//!
17//! Let `{$F$}` be the function mapping a [`Fork`] to all [`Fork`]s reachable from it,
18//! with `{$F(f)$}` a shorthand for ``{$F(#`Some`;(f))$}``:
19//! ```typm-render
20//!   F(#`None`)    &= {}\
21//!   F(#`Some`;(f)) &= { f } union
22//!     F(#`Some`;(f.#`branch`?.#`b_if`)) union
23//!     F(f.#`branch`?.#`b_else`).
24//! ```
25//!
26//! Let `{$L$}` be the function mapping a [`Fork`] to its leaves:
27//! ```typm-render
28//! L(f) = { g in F(f) | #`is_root`;(g) }.
29//! ```
30//!
31//! These definitions are used in pre-/post-conditions and invariants for
32//! functions and datatypes.
33

44/// An if/else branch. Invariant `{$I_b$}` declared here.
45///
46/// # Invariant `{$I_b$}`
47/// ```typm-render
48/// [exists #`b_else` . #`branch`.#`b_else` = #`Some`;(#`b_else`)]\
49/// ==> #`b_else`.#`entry` = #`branch`.#`b_if`.#`entry`.
50/// ```
51#[derive(Debug, Clone)]
52struct Branch {
53    condition: Vec<InternalInstruction>,
54    b_if: Rc<Fork>,
55    /// `None` indicates an unreachable branch
56    b_else: Option<Rc<Fork>>,
57}
58
59impl Branch {
60    fn collapse_with_returns(
61        &self,
62        func: &StepFunc,
63        wasm: &mut Vec<InternalInstruction>,
64        returns: &Vec<ValType>,
65    ) -> HQResult<()> {
66        wasm.extend(self.condition.iter().cloned());
67
68        let block_type = func.registries().types().function(
69            self.b_if
70                .entry
71                .clone()
72                .map(WasmProject::ir_type_to_wasm)
73                .collect::<Box<[_]>>()
74                .iter()
75                .copied()
76                .rev()
77                .collect_vec(),
78            returns.clone(),
79        )?;
80
81        wasm.push(InternalInstruction::Immediate(WInstruction::If(
82            BlockType::FunctionType(block_type),
83        )));
84
85        self.b_if.collapse_with_returns(func, wasm, returns)?;
86        wasm.push(InternalInstruction::Immediate(WInstruction::Else));
87
88        if let Some(b_else) = &self.b_else {
89            b_else.collapse_with_returns(func, wasm, returns)?;
90        } else {
91            wasm.push(InternalInstruction::Immediate(WInstruction::Unreachable));
92        }
93
94        wasm.push(InternalInstruction::Immediate(WInstruction::End));
95
96        Ok(())
97    }
98
99    /// Recursively collapse the branch into a list of instructions
100    fn collapse(&self, func: &StepFunc, wasm: &mut Vec<InternalInstruction>) -> HQResult<()> {
101        self.collapse_with_returns(func, wasm, &vec![])
102    }
103}

105/// A block of instructions, possibly ending in a branch, along with information
106/// about the type stack at the beginning of the block, and just before the branch
107#[derive(Debug, Clone)]
108struct Fork {
109    body: RefCell<Vec<InternalInstruction>>,
110    branch: RefCell<Option<Branch>>,
111    /// the type stack at the start of the body
112    entry: Rc<TypeStack>,
113    /// the type stack at the end of the body, just before any branches
114    exit: RefCell<Rc<TypeStack>>,
115}
116
117impl Fork {
118    fn new(type_stack: &Rc<TypeStack>) -> Rc<Self> {
119        Rc::new(Self {
120            body: RefCell::new(vec![]),
121            branch: RefCell::new(None),
122            entry: Rc::clone(type_stack),
123            exit: RefCell::new(Rc::clone(type_stack)),
124        })
125    }
126
127    /// Recursively collapse the fork into a list of instructions
128    fn collapse(&self, func: &StepFunc, wasm: &mut Vec<InternalInstruction>) -> HQResult<()> {
129        self.collapse_with_returns(func, wasm, &vec![])
130    }
131
132    fn collapse_with_returns(
133        &self,
134        func: &StepFunc,
135        wasm: &mut Vec<InternalInstruction>,
136        returns: &Vec<ValType>,
137    ) -> HQResult<()> {
138        wasm.extend(self.body.try_borrow()?.iter().cloned());
139
140        if let Some(branch) = &*self.branch.try_borrow()? {
141            branch.collapse_with_returns(func, wasm, returns)?;
142        }
143
144        Ok(())
145    }
146}

148/// Returns the NaN-box bit pattern for the provided type (which must be basic)
149fn boxed_pattern(ty: IrType) -> HQResult<i64> {
150    Ok(match ty {
151        IrType::Int => BOXED_INT_PATTERN,
152        IrType::Boolean => BOXED_BOOL_PATTERN,
153        IrType::String => BOXED_STRING_PATTERN,
154        IrType::ColorARGB => BOXED_COLOR_ARGB_PATTERN,
155        IrType::ColorRGB => BOXED_COLOR_RGB_PATTERN,
156        _ => hq_bug!("bad type for boxed pattern"),
157    })
158}

160/// Returns the list of instructions used to unbox a boxed value of the provided (base) type
161fn unbox_instructions(ty: IrType, func: &StepFunc) -> HQResult<Vec<InternalInstruction>> {
162    Ok(match ty {
163        IrType::Int | IrType::Boolean | IrType::ColorARGB | IrType::ColorRGB => wasm![I32WrapI64],
164        IrType::String => {
165            let table_index = func.registries().tables().register::<StringsTable, _>()?;
166            wasm![I32WrapI64, TableGet(table_index)]
167        }
168        IrType::Float => wasm![F64ReinterpretI64],
169        _ => hq_bug!("bad type for unboxing instructions"),
170    })
171}

173/// Returns the list of instructions used to check if a boxed value is of the specified (base) type
174fn box_type_check(ty: IrType) -> HQResult<Vec<InternalInstruction>> {
175    Ok(if ty == IrType::Float {
176        // float always comes last, and any i64 is always valid to be reinterpreted
177        // as an f64, so return true for these. The redundant branch will be
178        // optimised away by wasm-opt.
179        wasm![Drop, I32Const(1)]
180    } else {
181        let box_pattern = boxed_pattern(ty)?;
182        wasm![I64Const(box_pattern), I64And, I64Const(box_pattern), I64Eq,]
183    })
184}

186type BoxedLocals = (Box<[IrType]>, Box<[u32]>);

188fn build_boxed_locals(func: &StepFunc, inputs: Rc<[IrType]>) -> HQResult<BoxedLocals> {
189    let boxed_onward: Box<[_]> = inputs
190        .iter()
191        .copied()
192        .skip_while(|ty| ty.is_base_type())
193        .collect();
194    let locals = boxed_onward
195        .iter()
196        .copied()
197        .map(WasmProject::ir_type_to_wasm)
198        .map(|w_ty| func.local(w_ty))
199        .collect::<HQResult<Box<[_]>>>()?;
200    Ok((boxed_onward, locals))
201}

203fn make_leaf(func: &StepFunc, leaf: &Rc<Fork>, new_ty: IrType, local: u32) -> HQResult<Rc<Fork>> {
204    let base_ty = new_ty
205        .base_type()
206        .ok_or_else(|| make_hq_bug!("got non-base-type `new_ty`"))?;
207    let leaf = Fork::new(&*leaf.exit.try_borrow()?);
208    leaf.exit.try_borrow_mut()?.push_mut(new_ty);
209    let mut body = leaf.body.try_borrow_mut()?;
210    body.push(InternalInstruction::Immediate(WInstruction::LocalGet(
211        local,
212    )));
213    body.extend(unbox_instructions(base_ty, func)?);
214    Ok(Rc::clone(&leaf))
215}

217/// Makes a new branch for type selection, using the `latest_fork` as the else
218/// branch, and a new leaf as the if branch. Adds the new leaf to `new_leaves`.
219///
220/// Precondition: `new_ty.is_base_type()`
221///
222/// [`{$I_b$}`](Branch#invariant-i_b) maintained because every new leaf in this set of branches has
223/// the same entry.
224fn make_branch(
225    new_ty: IrType,
226    local: u32,
227    latest_fork: Option<&Rc<Fork>>,
228    new_leaf: &Rc<Fork>,
229) -> HQResult<Branch> {
230    let branch = Branch {
231        // TODO: condition
232        condition: wasm![LocalGet(local),]
233            .into_iter()
234            .chain(box_type_check(
235                #[expect(clippy::unwrap_used, reason = "guaranteed `Some` by precondition")]
236                new_ty.base_type().unwrap(),
237            )?)
238            .collect(),
239        b_if: Rc::clone(new_leaf),
240        b_else: latest_fork.cloned(),
241    };
242    Ok(branch)
243}

245/// Precondition: `new_ty.is_base_type()`
246/// Postcondition: `latest_fork.is_some()`
247fn make_branch_fork(
248    leaf: &Rc<Fork>,
249    new_ty: IrType,
250    local: u32,
251    latest_fork: &mut Option<Rc<Fork>>,
252    new_leaf: &Rc<Fork>,
253) -> HQResult<()> {
254    let branch = make_branch(new_ty, local, latest_fork.as_ref(), new_leaf)?;
255    let fork = Fork::new(&Rc::clone(&*leaf.exit.try_borrow()?));
256    *fork.branch.try_borrow_mut()? = Some(branch);
257    *latest_fork = Some(fork);
258    Ok(())
259}

261/// Postcondition: `new_leaves.is_some()`
262fn make_type_branches(
263    func: &StepFunc,
264    leaf: &Rc<Fork>,
265    input_ty: IrType,
266    base_tys: &[IrType],
267    local: u32,
268    new_leaves: &mut Option<Leaves>,
269) -> HQResult<()> {
270    let mut latest_fork = None;
271    let mut these_new_leaves =
272        Leaves::new(make_leaf(func, leaf, base_tys[0].and(input_ty), local)?);
273    for base_ty in base_tys.iter().skip(1).rev() {
274        these_new_leaves.push_leaf(make_leaf(func, leaf, base_ty.and(input_ty), local)?);
275    }
276    for (new_leaf, base_ty) in (&these_new_leaves)
277        .into_iter()
278        .skip(1)
279        .zip(base_tys.iter().skip(1).rev())
280    {
281        make_branch_fork(
282            leaf,
283            // precondition satisfied because `and` is nonincreasing
284            base_ty.and(input_ty),
285            local,
286            &mut latest_fork,
287            new_leaf,
288        )?;
289    }
290    let branch = make_branch(
291        // precondition satisfied because `and` is nonincreasing
292        base_tys[0].and(input_ty),
293        local,
294        latest_fork.as_ref(),
295        these_new_leaves.first(),
296    )?;
297    *leaf.branch.try_borrow_mut()? = Some(branch);
298    // `{$I_l$}` maintained because every new `Fork` in `these_new_leaves`
299    // has exactly one more type on the stack than the `exit` of
300    // the parent; inductively, `{$I_f'$}` holds for `new_leaves`.
301    // `{$I_f'$}` evident from creation of new leaves.
302    if let Some(new_leaves) = new_leaves {
303        new_leaves.extend(these_new_leaves);
304    } else {
305        *new_leaves = Some(these_new_leaves);
306    }
307
308    Ok(())
309}

311fn request_screen_refresh(
312    func: &StepFunc,
313    opcode: &IrOpcode,
314    mut body: RefMut<'_, Vec<InternalInstruction>>,
315) -> HQResult<()> {
316    if opcode.requests_screen_refresh() {
317        let refresh_requested = func.registries().globals().register(
318            "requests_refresh".into(),
319            (
320                ValType::I32,
321                ConstExpr::i32_const(0),
322                GlobalMutable(true),
323                GlobalExportable(true),
324            ),
325        )?;
326
327        body.append(&mut wasm![I32Const(1), #LazyGlobalSet(refresh_requested),]);
328    }
329    Ok(())
330}

332/// Should arguments to this opcode be unboxed?
333const fn should_unbox(opcode: &IrOpcode) -> bool {
334    // we don't want to unbox inputs to procedures, because... reasons,
335    // and we don't want to unbox inputs to `dup` or `swap` either because
336    // these are lower level instructions and don't really care about types
337    !matches!(
338        opcode,
339        &IrOpcode::procedures_call_warp(_) | &IrOpcode::hq_dup | &IrOpcode::hq_swap
340    )
341}

343/// Base, top-level branches across the whole function; i.e. branches that begin
344/// with an empty type stack. Invariants `{$I_r$}` and `{$I_r'$}` declared here.
345///
346/// # Invariant `{$I_r$}`
347/// ```typm-render
348/// forall #`root` in #`roots`;[1..#`roots`.#`len`;()) . #`root`.#`entry`.#`is_nil`;().
349/// ```
350///
351/// # Invariant `{$I_r'$}`
352/// ```typm-render
353///   forall #`root` in #`roots`[0..(#`roots`.#`len`() - 1)) .\
354///     forall #`leaf` in L(#`root`) . #`leaf`.#`exit`.#`is_nil`(),
355/// ```
356/// i.e. all except the last root must have all leaves with an empty type stack.
357struct Roots(Vec<Rc<Fork>>);
358
359impl Roots {
360    const fn new() -> Self {
361        Self(vec![])
362    }
363
364    /// Appends a root to the end of the sequence.
365    ///
366    /// Precondition: `is_root(root)`
367    fn push_root(&mut self, root: Rc<Fork>) {
368        self.0.push(root);
369    }
370}
371
372impl IntoIterator for Roots {
373    type Item = Rc<Fork>;
374    type IntoIter = alloc::vec::IntoIter<Rc<Fork>>;
375
376    fn into_iter(self) -> Self::IntoIter {
377        self.0.into_iter()
378    }
379}

381/// A collection of forks that are leaves. Invariant `{$I_l$}` declared here.
382///
383/// # Invariant `{$I_l$}`
384/// ```typm-render
385/// |{ #`leaf`.#`exit`.#`len`() | #`leaf` in #`leaves` }| = 1,
386/// ```
387/// i.e. all leaves have the same number of types on the stack.
388///
389/// # Invariant `{$I_l'$}`
390/// `!leaves.0.is_empty()`
391///
392/// This is maintained because all mutating methods either initialise the collection
393/// with a single leaf, or increase the length.
394///
395/// The invariant that all members of this sequence _are_ leaves is implied
396/// by invariant [`{$I_f'$}`](ForestBuilderState#invariant-i_f-1).
397struct Leaves(Vec<Rc<Fork>>);
398
399impl Leaves {
400    /// Creates a new collection of leaves, with one leaf to initialise.
401    ///
402    /// Precondition: `leaf.branch.is_none()`.
403    fn new(leaf: Rc<Fork>) -> Self {
404        Self(vec![leaf])
405    }
406
407    /// Reinitialises the collection with a single leaf.
408    ///
409    /// Precondition: `leaf.branch.is_none()`
410    fn reinit(&mut self, leaf: Rc<Fork>) {
411        self.0.clear();
412        self.push_leaf(leaf);
413    }
414
415    /// Appends a leaf to the end of the collection.
416    ///
417    /// Precondition: `leaf.branch.is_none()`
418    fn push_leaf(&mut self, leaf: Rc<Fork>) {
419        self.0.push(leaf);
420    }
421
422    /// Returns the first leaf in the collection.
423    ///
424    /// This is guaranteed to exist because .
425    fn first(&self) -> &Rc<Fork> {
426        &self.0[0]
427    }
428
429    /// The number of leaves in the collection.
430    const fn size(&self) -> usize {
431        self.0.len()
432    }
433
434    /// Replaces this collection of leaves with some other collection of leaves.
435    fn replace_with(&mut self, other: Self) {
436        self.0 = other.0;
437    }
438
439    /// Appends another collection of leaves onto this one
440    fn extend(&mut self, other: Self) {
441        self.0.extend(other.0);
442    }
443}

445impl<'a> IntoIterator for &'a Leaves {
446    type Item = &'a Rc<Fork>;
447    type IntoIter = core::slice::Iter<'a, Rc<Fork>>;
448
449    fn into_iter(self) -> Self::IntoIter {
450        self.0.iter()
451    }
452}

454/// State during [`build_forest`]. Invariants `{$I_f$}` and `{$I_f'$}` declared here.
455///
456/// # Invariant `{$I_f$}`
457/// ```typm-render
458/// #`roots`;[#`roots`.#`len`;() - 1] = #`this_root`.
459/// ```
460///
461/// # Invariant `{$I_f'$}`
462/// ```typm-render
463/// #`leaves` = L(#`this_root`).
464/// ```
465struct ForestBuilderState {
466    this_root: Rc<Fork>,
467    roots: Roots,
468    leaves: Leaves,
469}
470
471impl ForestBuilderState {
472    /// Adds a new root if the exit type stack on the latest tree is empty.
473    ///
474    /// Maintains invariant [`{$I_r$}`](Roots#invariant-i_r): a new root can
475    /// only be added if the previous root has all leaves with empty type stacks.
476    /// Inductively [`{$I_r$}`](Roots#invariant-i_r) must hold for any value of `roots`.
477    fn add_new_root_if_needed(&mut self) {
478        // By `{$I_l$}`, we only need to check if one leaf has an empty exit type stack.
479        // leaves[0] is guaranteed to exist by `{$I_f'$}`; since the tree structure is
480        // finite, there must exist at least one leaf.
481        if self
482            .leaves
483            .first()
484            .exit
485            .try_borrow()
486            .is_ok_and(|ty| ty.is_nil())
487        {
488            // Maintains `{$I_r'$}`
489            self.this_root = Fork::new(&Rc::new(TypeStack::Nil));
490            // Maintains `{$I_f$}`
491            self.roots.push_root(Rc::clone(&self.this_root));
492            // Maintains `{$I_f'$}`; this root is its own leaf. `{$I_l$}` trivial.
493            self.leaves.reinit(Rc::clone(&self.this_root));
494        }
495    }
496}
497
498impl ForestBuilderState {
499    fn new(type_stack: Rc<[IrType]>) -> Self {
500        let this_root = Fork::new(&type_stack.iter().copied().collect());
501        let mut roots = Roots::new();
502        // Initialise to one empty branch so that `{$I_r$}` and `{$I_r'$}` hold.
503        roots.push_root(Rc::clone(&this_root));
504        // $ L(`#this_root`) = { #`this_root` } $
505        let leaves = Leaves::new(Rc::clone(&this_root));
506        Self {
507            this_root,
508            roots,
509            leaves,
510        }
511    }
512}

514/// If all of the leaves have the same exit stack, we can pre-emptively merge
515/// them together to avoid a huge blow up in the number of leaves.
516///
517/// Precondition: $ #`leaves` = L(#`ancestor`) $
518fn collapse_same_typed_leaves(
519    func: &StepFunc,
520    ancestor: Rc<Fork>,
521    leaves: &mut Leaves,
522) -> HQResult<()> {
523    if leaves
524        .0
525        .iter()
526        .map(|new_leaf| Rc::clone(&*new_leaf.exit.borrow()))
527        .all_equal()
528    {
529        let mut collapsed = wasm![];
530        let same_exit = { Rc::clone(&*leaves.first().exit.try_borrow()?) };
531        ancestor.collapse_with_returns(
532            func,
533            &mut collapsed,
534            &same_exit
535                .clone()
536                .map(WasmProject::ir_type_to_wasm)
537                .collect::<Box<[_]>>()
538                .iter()
539                .copied()
540                .rev()
541                .collect_vec(),
542        )?;
543        *ancestor.body.try_borrow_mut()? = collapsed;
544        *ancestor.branch.try_borrow_mut()? = None;
545        *ancestor.exit.try_borrow_mut()? = same_exit;
546        leaves.reinit(Rc::clone(&ancestor));
547    }
548    Ok(())
549}

551/// Builds a 'forest' (collection of trees, in this case ordered) representing
552/// the WASM for the specified opcodes, with the entry point having the specified
553/// type stack.
554///
555/// Precondition: the specified type stack and opcodes are well-formed, i.e. all
556/// inputs are valid for all opcodes at all points.
557///
558/// Postcondition: returns `roots` s.t.
559/// ```typm-render
560///   forall &#`root` in #`roots` . #`is_root`;(#`root`) \
561///   and forall &#`root` in #`roots`;lr([0 .. (#`roots`.#`len`;() - 1))) space .\
562///       &forall #`leaf` in L(#`root`) . #`leaf`.#`exit`.#`is_nil`;().
563/// ```
564fn build_forest(
565    func: &StepFunc,
566    type_stack: Rc<[IrType]>,
567    opcodes: &[IrOpcode],
568) -> HQResult<Roots> {
569    let mut state = ForestBuilderState::new(type_stack);
570
571    for opcode in opcodes {
572        state.add_new_root_if_needed();
573
574        let expected_inputs = opcode.acceptable_inputs()?;
575
576        let mut new_leaves: Option<Leaves> = None;
577
578        for leaf in &state.leaves {
579            let actual_inputs = {
580                leaf.exit
581                    .try_borrow()?
582                    .clone()
583                    .take(expected_inputs.len())
584                    .collect::<Box<[_]>>()
585                    .iter()
586                    .copied()
587                    .rev()
588                    .collect::<Rc<[_]>>()
589            };
590
591            hq_assert_eq!(actual_inputs.len(), expected_inputs.len());
592
593            // All the inputs past the first boxed input (possibly empty)
594            let (boxed_onward, locals) = if should_unbox(opcode) {
595                build_boxed_locals(func, Rc::clone(&actual_inputs))?
596            } else {
597                (Box::from(&[] as &[_]), Box::from(&[] as &[_]))
598            };
599
600            // Block necessary here so that mutable references are dropped
601            {
602                // Store all inputs, from the first boxed one, in locals
603                leaf.body.try_borrow_mut()?.extend(
604                    locals
605                        .iter()
606                        .rev()
607                        .copied()
608                        .map(WInstruction::LocalSet)
609                        .map(InternalInstruction::Immediate),
610                );
611
612                let mut exit = leaf.exit.try_borrow_mut()?;
613                // `{$I_l$}` maintained because all leaves have exactly one element dropped
614                // from the type stack
615                exit.drop_mut(boxed_onward.len());
616            }
617
618            let mut these_new_leaves = Leaves::new(Rc::clone(leaf));
619
620            for (input_ty, local) in boxed_onward.iter().zip(locals) {
621                if input_ty.is_base_type() {
622                    for leaf_leaf in &these_new_leaves {
623                        leaf_leaf
624                            .body
625                            .try_borrow_mut()?
626                            .push(InternalInstruction::Immediate(WInstruction::LocalGet(
627                                local,
628                            )));
629                        // `{$I_l$}` maintained because exactly one element pushed for every leaf
630                        leaf_leaf.exit.try_borrow_mut()?.push_mut(*input_ty);
631                    }
632                } else {
633                    let base_tys: Box<[_]> = input_ty.base_types().collect();
634                    let mut these_these_new_leaves = None;
635                    for leaf_leaf in &these_new_leaves {
636                        make_type_branches(
637                            func,
638                            leaf_leaf,
639                            *input_ty,
640                            &base_tys,
641                            local,
642                            &mut these_these_new_leaves,
643                        )?;
644                    }
645
646                    these_new_leaves = #[expect(
647                        clippy::unwrap_used,
648                        reason = "postcondition of `make_type_branches` guarantees that this is \
649                                  Some"
650                    )]
651                    these_these_new_leaves.unwrap();
652                }
653                func.free_local(local)?;
654            }
655
656            for leaf_leaf in &these_new_leaves {
657                let mut exit = leaf_leaf.exit.try_borrow_mut()?;
658                // Remove top elements from stack. Maintains `{$I_l$}` because same action
659                // taken for every leaf.
660                let this_inputs: Rc<[_]> = exit
661                    .take_n(expected_inputs.len())
662                    .into_iter()
663                    .rev()
664                    .collect();
665                if this_inputs.len() != expected_inputs.len() {
666                    hq_bug!("not enough types on stack!");
667                }
668
669                let mut body = leaf_leaf.body.try_borrow_mut()?;
670                body.extend(opcode.wasm(func, Rc::clone(&this_inputs))?);
671
672                request_screen_refresh(func, opcode, body)?;
673
674                match opcode.output_type(this_inputs)? {
675                    ReturnType::None => (),
676                    ReturnType::Singleton(ty) => exit.push_mut(ty),
677                    ReturnType::MultiValue(tys) => {
678                        for ty in tys.iter() {
679                            exit.push_mut(*ty);
680                        }
681                    }
682                }
683            }
684
685            collapse_same_typed_leaves(func, Rc::clone(leaf), &mut these_new_leaves)?;
686
687            if let Some(new_leaves) = new_leaves.as_mut() {
688                new_leaves.extend(these_new_leaves);
689            } else {
690                new_leaves = Some(these_new_leaves);
691            }
692        }
693
694        state.leaves = #[expect(
695            clippy::unwrap_used,
696            reason = "`Leaves` always nonempty, so loop always runs at east once, so `new_leaves` \
697                      is definitely initialised to a `Some` value"
698        )]
699        new_leaves.unwrap();
700
701        collapse_same_typed_leaves(func, Rc::clone(&state.this_root), &mut state.leaves)?;
702    }
703
704    Ok(state.roots)
705}
706
707pub fn wrap_instructions(
708    func: &StepFunc,
709    type_stack: Rc<[IrType]>,
710    opcodes: &[IrOpcode],
711) -> HQResult<Vec<InternalInstruction>> {
712    let roots = build_forest(func, type_stack, opcodes)?;
713
714    let mut wasm = vec![];
715    for root in roots {
716        root.collapse(func, &mut wasm)?;
717    }
718
719    Ok(wasm)
720}

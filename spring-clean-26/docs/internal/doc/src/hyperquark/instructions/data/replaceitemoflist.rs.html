25pub fn wasm(
26    func: &StepFunc,
27    inputs: Rc<[IrType]>,
28    fields: &Fields,
29) -> HQResult<Vec<InternalInstruction>> {
30    hq_assert!(inputs.len() == 2);
31    let t1 = inputs[0];
32    let t2 = inputs[1];
33    let (list_global, maybe_length_global) = func.registries().lists().register(&fields.list)?;
34    let array_type = func.registries().lists().array_type(&fields.list)?;
35    let index_local = func.local(WasmProject::ir_type_to_wasm(t1))?;
36    let val_local = func.local(WasmProject::ir_type_to_wasm(t2))?;
37    func.free_local(index_local)?;
38    func.free_local(val_local)?;
39    Ok(wasm![
40        LocalSet(val_local),
41        LocalSet(index_local),
42        Block(WasmBlockType::Empty),
43        LocalGet(index_local),
44        I32Const(0),
45        I32LeS,
46        BrIf(0),
47        LocalGet(index_local),
48    ]
49    .into_iter()
50    .chain(if let Some(length_global) = maybe_length_global {
51        wasm![#LazyGlobalGet(length_global)]
52    } else {
53        let array_length = fields
54            .list
55            .initial_value()
56            .len()
57            .try_into()
58            .map_err(|_| make_hq_bug!("list initial value length out of bounds"))?;
59        wasm![I32Const(array_length)]
60    })
61    .chain(wasm![
62        I32GtS,
63        BrIf(0),
64        #LazyGlobalGet(list_global),
65        LocalGet(index_local),
66        I32Const(1),
67        I32Sub,
68        LocalGet(val_local),
69    ])
70    .chain(if fields.list.possible_types().is_base_type() {
71        vec![]
72    } else {
73        wasm![@boxed(t2)]
74    })
75    .chain(wasm![ArraySet(array_type), End,])
76    .collect())
77}

79pub fn acceptable_inputs(Fields { list }: &Fields) -> HQResult<Rc<[IrType]>> {

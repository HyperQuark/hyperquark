9fn input_names(block_info: &BlockInfo, context: &StepContext) -> HQResult<Vec<String>> {
10    let opcode = &block_info.opcode;
11    // target and procs need to be declared outside of the match block
12    // to prevent lifetime issues
13    let target = context.target();
14    let procs = target.procedures()?;
15    Ok(
16        #[expect(
17            clippy::wildcard_enum_match_arm,
18            reason = "too many opcodes to match individually"
19        )]
20        match opcode {
21            BlockOpcode::looks_say | BlockOpcode::looks_think => vec!["MESSAGE"],
22            BlockOpcode::operator_add
23            | BlockOpcode::operator_divide
24            | BlockOpcode::operator_subtract
25            | BlockOpcode::operator_multiply
26            | BlockOpcode::operator_mod => vec!["NUM1", "NUM2"],
27            BlockOpcode::operator_mathop | BlockOpcode::operator_round => vec!["NUM"],
28            BlockOpcode::operator_lt
29            | BlockOpcode::operator_gt
30            | BlockOpcode::operator_equals
31            | BlockOpcode::operator_and
32            | BlockOpcode::operator_or => vec!["OPERAND1", "OPERAND2"],
33            BlockOpcode::operator_join | BlockOpcode::operator_contains => {
34                vec!["STRING1", "STRING2"]
35            }
36            BlockOpcode::operator_letter_of => vec!["LETTER", "STRING"],
37            BlockOpcode::motion_gotoxy => vec!["X", "Y"],
38            BlockOpcode::motion_movesteps => vec!["STEPS"],
39            BlockOpcode::motion_pointindirection => vec!["DIRECTION"],
40            BlockOpcode::motion_turnleft | BlockOpcode::motion_turnright => vec!["DEGREES"],
41            BlockOpcode::sensing_keypressed => vec!["KEY_OPTION"],
42            BlockOpcode::sensing_dayssince2000
43            | BlockOpcode::data_variable
44            | BlockOpcode::argument_reporter_boolean
45            | BlockOpcode::argument_reporter_string_number
46            | BlockOpcode::looks_costume
47            | BlockOpcode::looks_size
48            | BlockOpcode::looks_nextcostume
49            | BlockOpcode::looks_costumenumbername
50            | BlockOpcode::looks_backdrops
51            | BlockOpcode::looks_hide
52            | BlockOpcode::looks_show
53            | BlockOpcode::pen_penDown
54            | BlockOpcode::pen_penUp
55            | BlockOpcode::pen_clear
56            | BlockOpcode::control_forever
57            | BlockOpcode::pen_menu_colorParam
58            | BlockOpcode::motion_direction
59            | BlockOpcode::data_deletealloflist
60            | BlockOpcode::data_lengthoflist
61            | BlockOpcode::data_listcontents
62            | BlockOpcode::control_stop
63            | BlockOpcode::event_broadcast_menu
64            | BlockOpcode::sensing_timer
65            | BlockOpcode::sensing_resettimer
66            | BlockOpcode::sensing_answer
67            | BlockOpcode::looks_backdropnumbername
68            | BlockOpcode::looks_nextbackdrop
69            | BlockOpcode::data_showvariable
70            | BlockOpcode::data_hidevariable
71            | BlockOpcode::sensing_mousex
72            | BlockOpcode::sensing_mousey
73            | BlockOpcode::sensing_mousedown
74            | BlockOpcode::motion_xposition
75            | BlockOpcode::motion_yposition
76            | BlockOpcode::sensing_keyoptions => vec![],
77            BlockOpcode::sensing_askandwait => vec!["QUESTION"],
78            BlockOpcode::event_broadcast | BlockOpcode::event_broadcastandwait => {
79                vec!["BROADCAST_INPUT"]
80            }
81            BlockOpcode::control_wait => vec!["DURATION"],
82            BlockOpcode::data_setvariableto
83            | BlockOpcode::data_changevariableby
84            | BlockOpcode::control_for_each => vec!["VALUE"],
85            BlockOpcode::operator_random => vec!["FROM", "TO"],
86            BlockOpcode::pen_setPenColorParamTo | BlockOpcode::pen_changePenColorParamBy => {
87                vec!["COLOR_PARAM", "VALUE"]
88            }
89            BlockOpcode::control_if
90            | BlockOpcode::control_if_else
91            | BlockOpcode::control_repeat_until
92            | BlockOpcode::control_while
93            | BlockOpcode::control_wait_until => vec!["CONDITION"],
94            BlockOpcode::operator_not => vec!["OPERAND"],
95            BlockOpcode::control_repeat => vec!["TIMES"],
96            BlockOpcode::operator_length => vec!["STRING"],
97            BlockOpcode::looks_switchcostumeto => vec!["COSTUME"],
98            BlockOpcode::looks_switchbackdropto => vec!["BACKDROP"],
99            BlockOpcode::looks_setsizeto | BlockOpcode::pen_setPenSizeTo => vec!["SIZE"],
100            BlockOpcode::looks_changesizeby => vec!["CHANGE"],
101            BlockOpcode::pen_setPenColorToColor => vec!["COLOR"],
102            BlockOpcode::data_addtolist
103            | BlockOpcode::data_itemnumoflist
104            | BlockOpcode::data_listcontainsitem => vec!["ITEM"],
105            BlockOpcode::data_itemoflist | BlockOpcode::data_deleteoflist => vec!["INDEX"],
106            BlockOpcode::data_replaceitemoflist | BlockOpcode::data_insertatlist => {
107                vec!["INDEX", "ITEM"]
108            }
109            BlockOpcode::motion_changexby => vec!["DX"],
110            BlockOpcode::motion_changeyby => vec!["DY"],
111            BlockOpcode::motion_setx => vec!["X"],
112            BlockOpcode::motion_sety => vec!["Y"],
113            BlockOpcode::procedures_call => 'proc_block: {
114                let serde_json::Value::String(proccode) = block_info
115                    .mutation
116                    .mutations
117                    .get("proccode")
118                    .ok_or_else(|| make_hq_bad_proj!("missing proccode on procedures_call"))?
119                else {
120                    hq_bad_proj!("non-string proccode on procedures_call");
121                };
122                let Some(proc) = procs.get(proccode.as_str()) else {
123                    break 'proc_block vec![];
124                };
125                proc.arg_ids().iter().map(|b| &**b).collect()
126            }
127            other => hq_todo!("unimplemented input_names for {:?}", other),
128        }
129        .into_iter()
130        .map(String::from)
131        .collect(),
132    )
133}

135pub fn inputs(
136    block_info: &BlockInfo,
137    blocks: &BlockMap,
138    context: &StepContext,
139    project: &Weak<IrProject>,
140    flags: &WasmFlags,
141) -> HQResult<Vec<IrOpcode>> {
142    Ok(input_names(block_info, context)?
143        .into_iter()
144        .map(|name| -> HQResult<Vec<IrOpcode>> {
145            let input = match block_info.inputs.get(&Box::from(&*name)) {
146                Some(noshadow @ Input::NoShadow(_, Some(_))) => noshadow,
147                Some(shadow @ Input::Shadow(_, Some(_), _)) => shadow,
148                None | Some(Input::NoShadow(_, None) | Input::Shadow(_, None, _)) => {
149                    // revert to a sensible default
150                    &Input::NoShadow(
151                        0,
152                        Some(BlockArrayOrId::Array(BlockArray::NumberOrAngle(6, 0.0))),
153                    )
154                }
155            };
156            #[expect(
157                clippy::wildcard_enum_match_arm,
158                reason = "all variants covered in previous match guards"
159            )]
160            match input {
161                Input::NoShadow(_, Some(block)) | Input::Shadow(_, Some(block), _) => match block {
162                    BlockArrayOrId::Array(arr) => {
163                        Ok(vec![from_special_block(arr, context, flags)?])
164                    }
165                    BlockArrayOrId::Id(id) => from_block(
166                        blocks.get(id).ok_or_else(|| {
167                            make_hq_bad_proj!("block for input {} doesn't exist", name)
168                        })?,
169                        blocks,
170                        context,
171                        project,
172                        NextBlocks::new(false),
173                        flags,
174                    ),
175                },
176                _ => hq_bad_proj!("missing input block for {}", name),
177            }
178        })
179        .collect::<HQResult<Vec<_>>>()?
180        .iter()
181        .flatten()
182        .cloned()
183        .collect())
184}

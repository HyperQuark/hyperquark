1use crate::instructions::{
2    DataSetvariabletoFields, DataTeevariableFields, DataVariableFields, HqCastFields, IrOpcode,
3};
4use crate::ir::{IrType, ReturnType};
5use crate::prelude::*;
6
7pub fn insert_casts(
8    blocks: &mut Vec<IrOpcode>,
9    ignore_variables: bool,
10    recurse: bool,
11) -> HQResult<()> {
12    let mut type_stack: Vec<(IrType, usize)> = vec![]; // a vector of types, and where they came from
13    let mut casts: Vec<(usize, IrType)> = vec![]; // a vector of cast targets, and where they're needed
14    for (i, block) in blocks.iter().enumerate() {
15        let mut expected_inputs = if ignore_variables
16            && (matches!(
17                block,
18                IrOpcode::data_setvariableto(DataSetvariabletoFields {
19                    local_write, ..
20                }) if !*local_write.borrow())
21                || matches!(block,
22                    IrOpcode::data_teevariable(DataTeevariableFields { local_read_write, .. }) if !*local_read_write.borrow())
23                || matches!(
24                    block,
25                    IrOpcode::data_addtolist(_) | IrOpcode::data_replaceitemoflist(_)
26                )) {
27            vec![IrType::Any]
28        } else {
29            block
30                .acceptable_inputs()?
31                .iter()
32                .copied()
33                .map(|ty| if ty.is_none() { IrType::Any } else { ty })
34                .collect::<Vec<_>>()
35        };
36        if type_stack.len() < expected_inputs.len() {
37            hq_bug!(
38                "didn't have enough inputs on the type stack\nat block {}",
39                block
40            );
41        }
42        let actual_inputs: Vec<_> = type_stack
43            .splice((type_stack.len() - expected_inputs.len()).., [])
44            .collect();
45        let mut dummy_actual_inputs: Vec<_> = actual_inputs.iter().map(|a| a.0).collect();
46        for (j, (expected, actual)) in
47            core::iter::zip(expected_inputs.clone(), actual_inputs).enumerate()
48        {
49            if !expected.is_none()
50                && !expected
51                    .base_types()
52                    .fold(IrType::none(), IrType::or)
53                    .contains(actual.0)
54            {
55                if matches!(
56                    block,
57                    IrOpcode::data_setvariableto(_)
58                        | IrOpcode::data_teevariable(_)
59                        | IrOpcode::data_addtolist(_)
60                        | IrOpcode::data_replaceitemoflist(_)
61                ) {
62                    hq_bug!(
63                        "attempted to insert a cast before a variable/list operation - variables \
64                         should encompass all possible types, rather than causing values to be \
65                         coerced.
66                        Tried to cast from {} (at position {}) to {} (at position {}).
67                        Occurred on these opcodes: [
68                        {}
69                        ]",
70                        actual.0,
71                        actual.1,
72                        expected,
73                        i,
74                        blocks.iter().map(|b| format!("{b}")).join(",\n"),
75                    )
76                }
77                casts.push((actual.1, expected));
78                dummy_actual_inputs[j] = expected;
79                expected_inputs[j] = IrOpcode::hq_cast(HqCastFields(expected))
80                    .output_type(Rc::from([if actual.0.is_none() {
81                        IrType::Any
82                    } else {
83                        actual.0
84                    }]))?
85                    .singleton_or_else(|| {
86                        make_hq_bug!("hq_cast returned no output type, or multiple output types")
87                    })?;
88            }
89        }
90        if ignore_variables
91            && (matches!(
92                block,
93                IrOpcode::data_variable(DataVariableFields {
94                    local_read, ..
95                }) if !*local_read.borrow())
96                || matches!(block,
97                    IrOpcode::data_teevariable(DataTeevariableFields { local_read_write, .. }) if !*local_read_write.borrow())
98                || matches!(
99                    block,
100                    IrOpcode::data_itemoflist(_) | IrOpcode::procedures_argument(_)
101                ))
102        {
103            type_stack.push((IrType::Any, i));
104        } else {
105            match block.output_type(Rc::from(dummy_actual_inputs))? {
106                ReturnType::Singleton(output) => type_stack.push((output, i)),
107                ReturnType::MultiValue(outputs) => {
108                    type_stack.extend(outputs.iter().copied().zip(core::iter::repeat(i)));
109                }
110                ReturnType::None => (),
111            }
112        }
113
114        if recurse && let Some(inline_steps) = block.inline_steps(false) {
115            for inline_step in inline_steps {
116                insert_casts(
117                    inline_step.try_borrow_mut()?.opcodes_mut(),
118                    ignore_variables,
119                    true,
120                )?;
121            }
122        }
123    }
124    for (pos, ty) in casts.into_iter().rev() {
125        blocks.insert(pos + 1, IrOpcode::hq_cast(HqCastFields(ty)));
126    }
127    Ok(())
128}

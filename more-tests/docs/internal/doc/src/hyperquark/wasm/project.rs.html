19/// A respresentation of a WASM representation of a project. Cannot be created directly;
20/// use `TryFrom<IrProject>`.
21pub struct WasmProject {

74    /// maps a broad IR type to a WASM type2
75    #[must_use]
76    pub fn ir_type_to_wasm(ir_type: IrType) -> ValType {
77        let base = ir_type.base_type();
78        match base {
79            Some(IrType::Float) => ValType::F64,
80            Some(IrType::Int | IrType::Boolean | IrType::ColorARGB | IrType::ColorRGB) => {
81                ValType::I32
82            }
83            Some(IrType::String) => ValType::EXTERNREF,
84            None => ValType::I64, // NaN boxed value... let's worry about colors later
85            Some(_) => unreachable!(),
86        }
87    }

89    pub fn finish(self) -> HQResult<FinishedWasm> {
90        let mut module = Module::new();
91
92        let mut memories = MemorySection::new();
93        let mut imports = ImportSection::new();
94        let mut types = TypeSection::new();
95        let mut functions = FunctionSection::new();
96        let mut codes = CodeSection::new();
97        let mut tables = TableSection::new();
98        let mut exports = ExportSection::new();
99        let mut elements = ElementSection::new();
100        let mut data = DataSection::new();
101        let mut globals = GlobalSection::new();
102
103        memories.memory(MemoryType {
104            minimum: 1,
105            maximum: None,
106            memory64: false,
107            shared: false,
108            page_size_log2: None,
109        });
110
111        let mut start_func = Function::new([]);
112
113        Rc::unwrap_or_clone(self.registries().tabled_strings().clone()).finish(
114            self.registries().strings(),
115            self.registries().tables().register::<StringsTable, _>()?,
116            &mut start_func,
117        )?;
118
119        Rc::unwrap_or_clone(self.registries().strings().clone()).finish(&mut imports);
120
121        self.registries().lists().clone().finish(
122            &mut data,
123            &mut elements,
124            &mut start_func,
125            self.imported_global_count()?,
126        )?;
127
128        start_func.instruction(&Instruction::End);
129
130        self.registries()
131            .external_functions()
132            .clone()
133            .finish(&mut imports, self.registries().types())?;
134
135        self.registries()
136            .static_functions()
137            .register_override::<SpawnNewThread, usize, _>((
138                self.registries().types().step_func_type()?,
139                self.registries().types().stack_struct_type()?,
140                self.registries().types().stack_array_type()?,
141                self.registries().types().thread_struct_type()?,
142                self.threads_table_index()?,
143            ))?;
144
145        self.registries()
146            .static_functions()
147            .register_override::<SpawnThreadInStack, usize, _>((
148                self.registries().types().step_func_type()?,
149                self.registries().types().stack_struct_type()?,
150                self.registries().types().stack_array_type()?,
151                self.registries().types().thread_struct_type()?,
152                self.threads_table_index()?,
153            ))?;
154
155        self.registries()
156            .static_functions()
157            .register_override::<MarkWaitingFlag, usize, _>(self.registries().types().struct_(
158                vec![FieldType {
159                    element_type: StorageType::I8,
160                    mutable: true,
161                }],
162            )?)?;
163
164        self.registries().static_functions().clone().finish(
165            &mut functions,
166            &mut exports,
167            &mut codes,
168            self.registries.types(),
169            self.imported_func_count()?,
170        )?;
171
172        for step_func in self.steps().try_borrow()?.iter().cloned() {
173            step_func.finish(
174                &mut functions,
175                &mut codes,
176                &self.events,
177                self.registries().types(),
178                self.threads_count_global()?,
179                self.spawn_new_thread_func()?,
180                self.spawn_thread_in_stack_func()?,
181                self.threads_table_index()?,
182                self.imported_func_count()?,
183                self.static_func_count()?,
184                self.imported_global_count()?,
185            )?;
186        }
187
188        self.tick_func(&mut functions, &mut codes, &mut exports)?;
189
190        self.finish_events(&mut functions, &mut codes, &mut exports)?;
191
192        self.unreachable_dbg_func(&mut functions, &mut codes, &mut exports)?;
193
194        codes.function(&start_func);
195        functions.function(self.registries().types().function(vec![], vec![])?);
196
197        let start_section = StartSection {
198            function_index: self.imported_func_count()? + functions.len() - 1,
199        };
200
201        self.registries()
202            .tables()
203            .register_override::<ThreadsTable, usize, _>(
204                self.registries().types().thread_struct_type()?,
205            )?;
206
207        elements.declared(Elements::Functions(
208            (self.imported_func_count()? + self.static_func_count()?
209                ..self.imported_func_count()?
210                    + self.static_func_count()?
211                    + u32::try_from(self.steps().try_borrow()?.len())
212                        .map_err(|_| make_hq_bug!("steps len out of bounds"))?)
213                .collect(),
214        ));
215
216        Rc::unwrap_or_clone(self.registries().types().clone()).finish(&mut types);
217
218        // TODO: make an elements registry to deal with this at the site of the block
219        for (table_index, table_name) in self
220            .registries()
221            .tables()
222            .registry()
223            .try_borrow()?
224            .keys()
225            .enumerate()
226        {
227            if table_name.starts_with("costume_names_") {
228                #[expect(
229                    clippy::unwrap_used,
230                    reason = "checked immediately before that this is a prefix of the table name"
231                )]
232                let target_index: u32 = table_name
233                    .strip_prefix("costume_names_")
234                    .unwrap()
235                    .parse()
236                    .map_err(|_| make_hq_bug!("couldn't parse target index from table name"))?;
237                let costume_names = self
238                    .costume_names
239                    .get(target_index as usize)
240                    .ok_or_else(|| make_hq_bug!("target index out of bounds for costume names"))?;
241                let name_globals = costume_names
242                    .iter()
243                    .map(|costume_name| {
244                        Ok(ConstExpr::global_get(
245                            u32::try_from(
246                                self.registries()
247                                    .strings()
248                                    .registry()
249                                    .try_borrow()?
250                                    .get_index_of(costume_name)
251                                    .ok_or_else(|| {
252                                        make_hq_bug!(
253                                            "couldn't find costume name string in strings registry"
254                                        )
255                                    })?,
256                            )
257                            .map_err(|_| make_hq_bug!("string index out of bounds"))?,
258                        ))
259                    })
260                    .collect::<HQResult<Box<[_]>>>()?;
261                elements.active(
262                    Some(
263                        table_index
264                            .try_into()
265                            .map_err(|_| make_hq_bug!("table index out of bounds"))?,
266                    ),
267                    &ConstExpr::i32_const(0),
268                    Elements::Expressions(RefType::EXTERNREF, Cow::Borrowed(&*name_globals)),
269                );
270            }
271        }
272
273        self.registries()
274            .tables()
275            .clone()
276            .finish(&mut tables, &mut exports);
277
278        exports.export("memory", ExportKind::Memory, 0);
279
280        self.registries().globals().clone().finish(
281            &mut globals,
282            &mut exports,
283            self.imported_global_count()?,
284            self.imported_func_count()?,
285            self.static_func_count()?,
286        );
287
288        module
289            .section(&types)
290            .section(&imports)
291            .section(&functions)
292            .section(&tables)
293            .section(&memories)
294            .section(&globals)
295            .section(&exports)
296            .section(&start_section)
297            .section(&elements)
298            .section(&DataCountSection { count: data.len() })
299            .section(&codes)
300            .section(&data);
301
302        let wasm_bytes = module.finish();
303
304        Ok(FinishedWasm {
305            wasm_bytes: wasm_bytes.into_boxed_slice(),
306            strings: self
307                .registries()
308                .strings()
309                .registry()
310                .borrow()
311                .keys()
312                .cloned()
313                .map(str::into_string)
314                .collect(),
315            target_names: self
316                .target_names
317                .into_iter()
318                .map(core::convert::Into::into)
319                .collect(),
320        })
321    }

417    #[expect(clippy::needless_pass_by_value, reason = "annoying to borrow a box")]
418    fn finish_event(
419        &self,
420        export_name: Box<str>,
421        indices: &[u32],
422        funcs: &mut FunctionSection,
423        codes: &mut CodeSection,
424        exports: &mut ExportSection,
425    ) -> HQResult<u32> {
426        let mut func = Function::new(vec![]);
427
428        let threads_count = self.threads_count_global()?;
429
430        let spawn_new_thread = self.spawn_new_thread_func()?;
431
432        let instrs = indices
433            .iter()
434            .map(|&i| {
435                Ok(wasm![
436                    RefFunc(i + self.imported_func_count()? + self.static_func_count()?),
437                    RefNull(HeapType::Abstract {
438                        shared: false,
439                        ty: AbstractHeapType::Struct
440                    }),
441                    #StaticFunctionCall(spawn_new_thread),
442                ])
443            })
444            .flatten_ok()
445            .collect::<HQResult<Vec<_>>>()?;
446
447        for instruction in instrs {
448            for real_instruction in instruction.eval(
449                &self.events,
450                self.registries().types(),
451                self.threads_count_global()?,
452                self.spawn_new_thread_func()?,
453                self.spawn_thread_in_stack_func()?,
454                self.threads_table_index()?,
455                self.imported_func_count()?,
456                self.static_func_count()?,
457                self.imported_global_count()?,
458            )? {
459                func.instruction(&real_instruction);
460            }
461        }
462        for instruction in wasm![
463            #LazyGlobalGet(threads_count),
464            I32Const(
465                i32::try_from(indices.len())
466                    .map_err(|_| make_hq_bug!("indices len out of bounds"))?
467            ),
468            I32Add,
469            #LazyGlobalSet(threads_count)
470        ] {
471            for real_instruction in instruction.eval(
472                &self.events,
473                self.registries().types(),
474                self.threads_count_global()?,
475                self.spawn_new_thread_func()?,
476                self.spawn_thread_in_stack_func()?,
477                self.threads_table_index()?,
478                self.imported_func_count()?,
479                self.static_func_count()?,
480                self.imported_global_count()?,
481            )? {
482                func.instruction(&real_instruction);
483            }
484        }
485        func.instruction(&Instruction::End);
486
487        funcs.function(self.registries().types().function(vec![], vec![])?);
488        codes.function(&func);
489        exports.export(
490            &export_name,
491            ExportKind::Func,
492            self.imported_func_count()? + funcs.len() - 1,
493        );
494
495        Ok(self.imported_func_count()? + funcs.len() - 1)
496    }

498    fn finish_events(
499        &self,
500        funcs: &mut FunctionSection,
501        codes: &mut CodeSection,
502        exports: &mut ExportSection,
503    ) -> HQResult<()> {
504        let event_funcs = self
505            .events
506            .iter()
507            .map(|(event, indices)| {
508                Ok(Some((
509                    event,
510                    self.finish_event(
511                        match event {
512                            Event::FlagClicked => "flag_clicked".into(),
513                            Event::Broadcast(_) => return Ok(None), // broadcasts handled in the sender blocks
514                            Event::SpriteClicked(index) => {
515                                format!("spriteClicked{index}").into_boxed_str()
516                            }
517                        },
518                        indices,
519                        funcs,
520                        codes,
521                        exports,
522                    )?,
523                )))
524            })
525            .collect::<HQResult<Box<[_]>>>()?
526            .into_iter()
527            .flatten()
528            .collect::<BTreeMap<_, _>>();
529        let sprite_clicked_indices: Box<[i32]> = self
530            .events
531            .keys()
532            .filter_map(|e| {
533                if let Event::SpriteClicked(index) = e {
534                    Some(
535                        i32::try_from(*index)
536                            .map_err(|_| make_hq_bug!("target index out of bounds")),
537                    )
538                } else {
539                    None
540                }
541            })
542            .collect::<HQResult<_>>()?;
543        if !sprite_clicked_indices.is_empty() {
544            let mut sprite_clicked_func = Function::new([]);
545            let sprite_clicked_instrs: Vec<_> = sprite_clicked_indices
546                .iter()
547                .flat_map(|index| {
548                    wasm![
549                        LocalGet(0),
550                        I32Const(*index),
551                        I32Eq,
552                        If(WasmBlockType::Empty),
553                        Call(
554                            event_funcs[&Event::SpriteClicked(
555                                #[expect(
556                                    clippy::unwrap_used,
557                                    reason = "guaranteed to succeed because i32 was originally a \
558                                              u32"
559                                )]
560                                (*index).try_into().unwrap()
561                            )]
562                        ),
563                        Return,
564                        End,
565                    ]
566                })
567                .chain(wasm![End])
568                .collect();
569            for instruction in sprite_clicked_instrs {
570                for real_instruction in instruction.eval(
571                    &self.events,
572                    self.registries().types(),
573                    self.threads_count_global()?,
574                    self.spawn_new_thread_func()?,
575                    self.spawn_thread_in_stack_func()?,
576                    self.threads_table_index()?,
577                    self.imported_func_count()?,
578                    self.static_func_count()?,
579                    self.imported_global_count()?,
580                )? {
581                    sprite_clicked_func.instruction(&real_instruction);
582                }
583            }
584            funcs.function(
585                self.registries()
586                    .types()
587                    .function(vec![ValType::I32], vec![])?,
588            );
589            codes.function(&sprite_clicked_func);
590            exports.export(
591                "trigger_sprite_clicked",
592                ExportKind::Func,
593                self.imported_func_count()? + funcs.len() - 1,
594            );
595        }
596
597        Ok(())
598    }

600    fn tick_func(
601        &self,
602        funcs: &mut FunctionSection,
603        codes: &mut CodeSection,
604        exports: &mut ExportSection,
605    ) -> HQResult<()> {
606        let thread_struct_type = self.registries().types().thread_struct_type()?;
607        let stack_struct_ty = self.registries().types().stack_struct_type()?;
608
609        let mut tick_func = Function::new(vec![
610            (2, ValType::I32),
611            (
612                1,
613                ValType::Ref(RefType {
614                    nullable: true,
615                    heap_type: HeapType::Concrete(thread_struct_type),
616                }),
617            ),
618            (
619                1,
620                ValType::Ref(RefType {
621                    nullable: false,
622                    heap_type: HeapType::Concrete(stack_struct_ty),
623                }),
624            ),
625        ]);
626
627        let step_func_ty = self.registries().types().step_func_type()?;
628        let stack_array_ty = self.registries().types().stack_array_type()?;
629
630        let instructions = wasm![
631            TableSize(self.threads_table_index()?),
632            LocalTee(1),
633            I32Eqz,
634            BrIf(0),
635            Loop(WasmBlockType::Empty),
636            LocalGet(0),
637            LocalGet(0),
638            TableGet(self.threads_table_index()?),
639            LocalTee(2),
640            RefIsNull,
641            If(WasmBlockType::Empty),
642            LocalGet(0),
643            I32Const(1),
644            I32Add,
645            LocalTee(0),
646            LocalGet(1),
647            I32LtS,
648            If(WasmBlockType::Empty),
649            Br(2),
650            Else,
651            Return,
652            End,
653            End,
654            LocalGet(2),
655            RefAsNonNull,
656            StructGet {
657                struct_type_index: thread_struct_type,
658                field_index: 1
659            },
660            LocalGet(2),
661            RefAsNonNull,
662            StructGet {
663                struct_type_index: thread_struct_type,
664                field_index: 0
665            },
666            I32Const(1),
667            I32Sub,
668            ArrayGet(stack_array_ty),
669            RefAsNonNull,
670            LocalTee(3),
671            StructGet {
672                struct_type_index: stack_struct_ty,
673                field_index: 1
674            },
675            LocalGet(3),
676            StructGet {
677                struct_type_index: stack_struct_ty,
678                field_index: 0
679            },
680            CallRef(step_func_ty),
681            LocalGet(0),
682            I32Const(1),
683            I32Add,
684            LocalTee(0),
685            LocalGet(1),
686            I32LtS,
687            BrIf(0),
688            End,
689        ];
690        for instr in instructions {
691            for real_instruction in instr.eval(
692                &self.events,
693                self.registries().types(),
694                self.threads_count_global()?,
695                self.spawn_new_thread_func()?,
696                self.spawn_thread_in_stack_func()?,
697                self.threads_table_index()?,
698                self.imported_func_count()?,
699                self.static_func_count()?,
700                self.imported_global_count()?,
701            )? {
702                tick_func.instruction(&real_instruction);
703            }
704        }
705        tick_func.instruction(&Instruction::End);
706        funcs.function(self.registries().types().function(vec![], vec![])?);
707        codes.function(&tick_func);
708        exports.export(
709            "tick",
710            ExportKind::Func,
711            funcs.len() + self.imported_func_count()? - 1,
712        );
713        Ok(())
714    }

716    pub fn from_ir(
717        ir_project: &Rc<IrProject>,
718        _ssa_token: crate::optimisation::SSAToken,
719        flags: WasmFlags,
720    ) -> HQResult<Self> {
721        let steps = Rc::new(RefCell::new(Vec::new()));
722        let registries = Rc::new(Registries::default());
723        let mut events: BTreeMap<Event, Vec<u32>> = BTreeMap::default();
724        events.insert(Event::FlagClicked, vec![]); // make sure that `flag_clicked` will be defined
725        let costume_names = Rc::new(
726            ir_project
727                .targets()
728                .try_borrow()?
729                .values()
730                .map(|target| {
731                    target
732                        .costumes()
733                        .iter()
734                        .map(|costume| costume.name.clone())
735                        .collect()
736                })
737                .collect(),
738        );
739        for (i, step) in ir_project.steps().try_borrow()?.iter().enumerate() {
740            crate::log!("compiling step {i}");
741            StepFunc::compile_step(
742                step,
743                StepIndex(i),
744                &steps,
745                Rc::clone(&registries),
746                flags,
747                Rc::clone(&costume_names),
748            )?;
749        }
750        // add thread event handlers for them
751        for thread in ir_project.threads().try_borrow()?.iter() {
752            events.entry(thread.event().clone()).or_default().push(
753                u32::try_from(thread.first_step().0)
754                    .map_err(|_| make_hq_bug!("step func index out of bounds"))?,
755            );
756        }
757        Ok(Self {
758            flags,
759            steps,
760            events,
761            registries,
762            environment: ExternalEnvironment::WebBrowser,
763            target_names: ir_project.targets().try_borrow()?.keys().cloned().collect(),
764            costume_names,
765        })
766    }
767}

1use core::ops::Deref;
2
3use crate::instructions::{
4    DataVariableFields, HqBoxFields, HqYieldFields, IrOpcode, ProceduresArgumentFields, YieldMode,
5};
6use crate::ir::{RcVar, Step};
7use crate::prelude::*;
8
9pub fn box_proc_returns<S>(
10    step: S,
11    rev_ret_vars: &[&RcVar],
12    visited_steps: &mut BTreeSet<Box<str>>,
13) -> HQResult<()>
14where
15    S: Deref<Target = RefCell<Step>>,
16{
17    if visited_steps.contains(step.try_borrow()?.id()) {
18        return Ok(());
19    }
20
21    visited_steps.insert(step.try_borrow()?.id().into());
22
23    let opcodes_len = step.try_borrow()?.opcodes().len();
24
25    for i in 0..opcodes_len {
26        let inlined_steps = step
27            .try_borrow()?
28            .opcodes()
29            .get(i)
30            .ok_or_else(|| make_hq_bug!("opcode index out of bounds"))?
31            .inline_steps(true);
32        if let Some(inline_steps) = inlined_steps {
33            let new_inline_steps = inline_steps
34                .iter()
35                .map(|inline_step| {
36                    let inline_step_mut = Rc::new(Rc::unwrap_or_clone(Rc::clone(inline_step)));
37                    box_proc_returns(Rc::clone(&inline_step_mut), rev_ret_vars, visited_steps)?;
38                    Ok(inline_step_mut)
39                })
40                .collect::<HQResult<Box<[_]>>>()?;
41            for (step_ref_mut, new_inline_step) in step
42                .try_borrow_mut()?
43                .opcodes_mut()
44                .get_mut(i)
45                .ok_or_else(|| make_hq_bug!("opcode index out of bounds"))?
46                .inline_steps_mut(true)
47                .ok_or_else(|| {
48                    make_hq_bug!("inline_steps_mut was None when inline_steps was Some")
49                })?
50                .iter_mut()
51                .zip(new_inline_steps)
52            {
53                **step_ref_mut = new_inline_step;
54            }
55        }
56    }
57
58    let mut box_additions = Vec::new();
59
60    let mut rev_vars_iter = rev_ret_vars.iter();
61
62    let mut to_skip = 0;
63
64    for (i, opcode) in step
65        .try_borrow()?
66        .opcodes()
67        .iter()
68        .enumerate()
69        .rev()
70        .filter(|(_, op)| {
71            !matches!(
72                op,
73                IrOpcode::hq_yield(HqYieldFields {
74                    mode: YieldMode::Return,
75                }),
76            )
77        })
78    {
79        #[expect(
80            clippy::wildcard_enum_match_arm,
81            reason = "too many variants to match explicitly"
82        )]
83        match opcode {
84            IrOpcode::data_variable(DataVariableFields { var, .. }) => {
85                if to_skip > 0 {
86                    to_skip -= 1;
87                    continue;
88                }
89                let Some(ret_var) = rev_vars_iter.next() else {
90                    break;
91                };
92                if var.borrow().possible_types().is_base_type()
93                    && !ret_var.possible_types().is_base_type()
94                {
95                    box_additions.push((i + 1, *ret_var.possible_types()));
96                }
97            }
98            IrOpcode::procedures_argument(ProceduresArgumentFields { arg_var, .. }) => {
99                if to_skip > 0 {
100                    to_skip -= 1;
101                    continue;
102                }
103                let Some(ret_var) = rev_vars_iter.next() else {
104                    break;
105                };
106                if arg_var.possible_types().is_base_type()
107                    && !ret_var.possible_types().is_base_type()
108                {
109                    box_additions.push((i + 1, *ret_var.possible_types()));
110                }
111            }
112            IrOpcode::data_setvariableto(_) => {
113                to_skip += 1;
114            }
115            _ => {
116                break;
117            }
118        }
119    }
120
121    let mut step_mut = step.try_borrow_mut()?;
122    let opcodes_mut = step_mut.opcodes_mut();
123
124    for (box_addition, output_ty) in box_additions {
125        opcodes_mut.splice(
126            box_addition..box_addition,
127            vec![IrOpcode::hq_box(HqBoxFields { output_ty })],
128        );
129    }
130
131    Ok(())
132}

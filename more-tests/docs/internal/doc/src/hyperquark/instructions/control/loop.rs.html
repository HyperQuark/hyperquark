8#[derive(Debug, Clone)]
9pub struct Fields {
10    pub first_condition: Option<Rc<RefCell<Step>>>,
11    pub condition: Rc<RefCell<Step>>,
12    pub body: Rc<RefCell<Step>>,
13    pub pre_body: Option<Rc<RefCell<Step>>>,
14    pub flip_if: bool,
15}

17impl fmt::Display for Fields {
18    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
19        write!(f, "{{")?;
20        if let Some(ref step) = self.first_condition {
21            write!(f, r#""first_condition": {},"#, RefCell::borrow(step))?;
22        }
23        if let Some(ref step) = self.pre_body {
24            write!(f, r#""pre_body": {},"#, RefCell::borrow(step))?;
25        }
26        write!(
27            f,
28            r#"
29        "condition": {},
30        "body": {},
31        "flip_if": {}
32    }}"#,
33            RefCell::borrow(&self.condition),
34            RefCell::borrow(&self.body),
35            self.flip_if
36        )
37    }
38}

40pub fn wasm(
41    func: &StepFunc,
42    _inputs: Rc<[IrType]>,
43    Fields {
44        first_condition,
45        condition,
46        body,
47        flip_if,
48        pre_body,
49    }: &Fields,
50) -> HQResult<Vec<InternalInstruction>> {
51    let inner_instructions = func.compile_inner_step(Rc::clone(body))?;
52    let first_condition_instructions = func.compile_inner_step(
53        first_condition
54            .clone()
55            .unwrap_or_else(|| Rc::clone(condition)),
56    )?;
57    let condition_instructions = func.compile_inner_step(Rc::clone(condition))?;
58    let pre_body_instructions = pre_body.as_ref().map_or_else(
59        || Ok(vec![]),
60        |pre_body_step| func.compile_inner_step(Rc::clone(pre_body_step)),
61    )?;
62    Ok(wasm![Block(BlockType::Empty),]
63        .into_iter()
64        .chain(first_condition_instructions)
65        .chain(if *flip_if {
66            wasm![BrIf(0), Loop(BlockType::Empty)]
67        } else {
68            wasm![I32Eqz, BrIf(0), Loop(BlockType::Empty)]
69        })
70        .chain(pre_body_instructions)
71        .chain(inner_instructions)
72        .chain(condition_instructions)
73        .chain(if *flip_if {
74            wasm![I32Eqz, BrIf(0), End, End]
75        } else {
76            wasm![BrIf(0), End, End]
77        })
78        .collect())
79}

117    fn make_condition_step(context: &StepContext) -> Rc<RefCell<Step>> {
118        Rc::new(RefCell::new(Step::new(
119            None,
120            context.clone(),
121            vec![crate::instructions::IrOpcode::hq_boolean(
122                crate::instructions::HqBooleanFields(true),
123            )],
124            Weak::new(),
125            false,
126        )))
127    }
128
129    pub fn make_fields(first_condition: bool, pre_body: bool, flip_if: bool) -> Fields {
130        let target = make_target();
131        let context = make_context(&target);
132        Fields {
133            first_condition: if first_condition {
134                None
135            } else {
136                Some(make_condition_step(&context))
137            },
138            condition: make_condition_step(&context),
139            pre_body: if pre_body {
140                None
141            } else {
142                Some(make_step(&target))
143            },
144            body: make_step(&target),
145            flip_if,
146        }
147    }
148}

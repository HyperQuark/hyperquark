26pub fn wasm(
27    func: &StepFunc,
28    _inputs: Rc<[IrType]>,
29    Fields {
30        branch_if,
31        branch_else,
32    }: &Fields,
33) -> HQResult<Vec<InternalInstruction>> {
34    let if_instructions = func.compile_inner_step(Rc::clone(branch_if))?;
35    let else_instructions = func.compile_inner_step(Rc::clone(branch_else))?;
36    let block_type = func
37        .registries()
38        .types()
39        .function(vec![ValType::I32], vec![])?;
40    Ok(wasm![
41        Block(BlockType::FunctionType(block_type)),
42        Block(BlockType::FunctionType(block_type)),
43        I32Eqz,
44        BrIf(0)
45    ]
46    .into_iter()
47    .chain(if_instructions)
48    .chain(wasm![Br(1), End,])
49    .chain(else_instructions)
50    .chain(wasm![End])
51    .collect())
52}

64pub fn const_fold(
65    inputs: &[ConstFoldItem],
66    _state: &mut ConstFoldState,
67    fields: &Fields,
68) -> HQResult<ConstFold> {
69    if let ConstFoldItem::Basic(VarVal::Bool(const_condition)) = inputs[0]
70        && false
71    {
72        Ok(ConstFold::Folded(Rc::from([ConstFoldItem::Stack(
73            RefCell::borrow(if const_condition {
74                &fields.branch_if
75            } else {
76                &fields.branch_else
77            })
78            .opcodes()
79            .iter()
80            .cloned()
81            .collect(),
82        )])))
83    } else {
84        Ok(NotFoldable)
85    }
86}

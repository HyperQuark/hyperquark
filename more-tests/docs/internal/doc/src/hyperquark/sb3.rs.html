262/// Possible fields (rectangular) in a block
263#[derive(Serialize, Deserialize, Debug, Clone, EnumFieldGetter, PartialEq)]
264#[serde(untagged)]
265pub enum Field {
266    Value((Option<VarVal>,)),
267    ValueId(Option<VarVal>, Option<Box<str>>),
268}
269
270impl Field {
271    // this isn't auto implemented by EnumFieldGetter because Field::Value is actually a tuple
272    // in a tuple, so that serde correctly parses a single-item array
273    #[must_use]
274    pub const fn get_0(&self) -> Option<&VarVal> {
275        match self {
276            Self::ValueId(val, _) | Self::Value((val,)) => val.as_ref(),
277        }
278    }
279}

281/// Represents a mutation on a block. See <https://en.scratch-wiki.info/wiki/Scratch_File_Format#Mutations>
282#[derive(Serialize, Deserialize, Debug, Clone, PartialEq, Eq)]
283#[serde(rename_all = "camelCase")]
284pub struct Mutation {
285    /// ignored - should always be "mutation"
286    pub tag_name: Box<str>,
287    /// ignored - should always be []
288    #[serde(default)]
289    pub children: Vec<()>,
290    #[serde(flatten)]
291    pub mutations: BTreeMap<Box<str>, Value>,
292}
293
294impl Default for Mutation {
295    fn default() -> Self {
296        Self {
297            tag_name: "mutation".into(),
298            children: vec![],
299            mutations: BTreeMap::new(),
300        }
301    }
302}

481
482    fn try_from(string: String) -> Result<Self, Self::Error> {
483        (&string[..]).try_into()
484    }
485}
486
487impl TryFrom<&str> for Sb3Project {
488    type Error = HQError;
489
490    fn try_from(string: &str) -> Result<Self, Self::Error> {
491        use serde_json::error::Category;
492        let sb3: Result<Self, serde_json::Error> = serde_json::from_str(string);
493        match sb3 {
494            Ok(proj) => Ok(proj),
495            Err(err) => match err.classify() {
496                Category::Syntax => hq_bad_proj!(
497                    "Invalid JSON syntax at project.json:{}:{}",
498                    err.line(),
499                    err.column()
500                ),
501                Category::Data => hq_bad_proj!(
502                    "Invalid project.json at project.json:{}:{}. Actual error: {}",
503                    err.line(),
504                    err.column(),
505                    err
506                ),
507                Category::Eof => hq_bad_proj!(
508                    "Unexpected end of file at project.json:{}:{}",
509                    err.line(),
510                    err.column()
511                ),
512                Category::Io => hq_bug!("Failed to deserialize json due to IO error"),
513            },
514        }
515    }
516}
